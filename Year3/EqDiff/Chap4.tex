\chapter{Conceptos básicos de estabilidad y estabilidad para sistemas lineales}
Estabilidad : si cambiando ligeramente las condiciones iniciales la solución cambia poco para todo tiempo $\displaystyle t \geq t_{0} $. 
\section{Conceptos de estabilidad}
Supongamos que $\displaystyle f : D \subset \R \times \R^{n} \to \R^{n} $ es una función continua y localmente Lipschitz respecto a la segunda variable. Sea $\displaystyle \overline{x} : [\tau, \infty) \to \R $ una solución del sistema $\displaystyle \overline{x}' = f\left(t, \overline{x}\right) $. 
\begin{definition}[Estabilidad]
Diremos que $\displaystyle \overline{x} $ es una solución \textbf{estable} si para todo $\displaystyle \epsilon > 0 $ y $\displaystyle t_{0}\geq \tau $ existe $\displaystyle \delta > 0 $ tal que si $\displaystyle \|x_{0}-\overline{x}\left(t_{0}\right)\| < \delta  $ entonces $\displaystyle \|\varphi\left(t; t_{0}, x_{0}\right)-\overline{x}\left(t\right)\| < \epsilon  $, $\displaystyle \forall t \geq t_{0} $. Diremos que una solución es \textbf{inestable} si no es estable.  
\end{definition}
\begin{observation}
Para que una solución sea estable tiene que estar definida en un intervalo $\displaystyle [\tau, \infty )$ para cierto $\displaystyle \tau \in [-\infty, \infty) $. 
\end{observation}
\begin{definition}[Estabilidad asintótica]
Diremos que $\displaystyle \overline{x} $ es una solución \textbf{asintóticamente estable} si es estable y para todo $\displaystyle t_{0}\geq \tau $ existe $\displaystyle \eta > 0 $ tal que si $\displaystyle \left\|x_{0}-\overline{x}\left(t_{0}\right)\right\|<\eta  $ entonces
\[ \left\|\varphi\left(t; t_{0}, x_{0}\right)-\overline{x}\left(t\right)\right\|\xrightarrow{t \to \infty} 0 .\]
\end{definition}
\begin{eg}
Consideremos la EDO $\displaystyle x' = ax $. Tendremos que el flujo se puede obtener explícitamente:
\[\varphi\left(t; t_{0}, x_{0}\right)=x_{0}e^{a\left(t-t_{0}\right)} .\]
Distinguimos varios casos
\begin{itemize}
\item Si $\displaystyle a < 0 $, toda solución es asintóticamente estable. En efecto, consideremos la solución 
	\[ \overline{x}\left(t\right)= \overline{x}\left(t_{0}\right)e^{a\left(t-t_{0}\right)} .\]
	Así, dado $\displaystyle \epsilon > 0 $ podemos tomar $\displaystyle \delta = \epsilon  $ y $\displaystyle x_{0} $ tal que $\displaystyle \left|x_{0} - \overline{x}\left(t_{0}\right)\right|< \epsilon  $, entonces
	\[ \left|\varphi\left(t; t_{0}, x_{0}\right)-\overline{x}\left(t\right)\right|= \left|x_{0}-\overline{x}\left(t_{0}\right)\right|e^{a\left(t-t_{0}\right)} \leq \left|x_{0}-\overline{x}\left(t_{0}\right)\right|<\epsilon  , \; \forall t \geq t_{0}.\]
	Así, tenemos que la solución es estable. Además, 
	\[\lim_{t \to \infty} \left|\varphi\left(t; t_{0}, x_{0}\right)-\overline{x}\left(t\right)\right| = \lim_{t \to \infty } \left|x_{0}-\overline{x}\left(t_{0}\right)\right|e^{a\left(t-t_{0}\right)} = 0 ,\]
	por lo que la solución es asintóticamente estable (tomando $\displaystyle \eta > 0 $ arbitrario).
\item Si $\displaystyle a = 0 $, las soluciones son estables pero no asintóticamente estables. Esto se sigue trivialmente del hecho de que en este caso toda solución es estable. 
\item Si $\displaystyle a > 0 $ toda solución es inestable. Independientemente de $\displaystyle x_{0} \in \R \backslash \left\{ \overline{x}\left(t_{0}\right)\right\}  $ se tiene que 
	\[\lim_{t \to \infty } \left|\varphi\left(t; t_{0}, x_{0}\right)-\overline{x}\left(t\right)\right| = \lim_{t \to \infty } \left|x_{0}-\overline{x}\left(t_{0}\right)\right|e^{a\left(t-t_{0}\right)} = \infty  .\]
\end{itemize}
\end{eg}
\begin{eg}
Consideremos la EDO no lineal $\displaystyle x ' = x^{2} $, cuyo flujo se puede obtener explícitamente:
\[\varphi\left(t; t_{0}, x_{0}\right)=\frac{x_{0}}{1-x_{0}\left(t-t_{0}\right)} .\]
En este caso la estabilidad depende de la solución cuya estabilidad queremos estudiar. Tomemos 
\[ \overline{x}\left(t\right)=\frac{\overline{x}\left(t_{0}\right)}{1-\overline{x}\left(t_{0}\right)\left(t-t_{0}\right)} .\]
Por un lado, si $\displaystyle \overline{x}\left(t_{0}\right) > 0 $ observamos que la solución sólo está definida en el intervalo $\displaystyle \left(-\infty, t_{0} + \frac{1}{x_{0}}\right) $. Como no está definida hasta $\displaystyle \infty  $, no es estable. Esto mismo hace imposible la estabilidad de la solución $\displaystyle \overline{x}\left(t\right)\equiv 0 $. En efecto, por muy cerca que elijamos $\displaystyle x_{0} $, si $\displaystyle x_{0} > 0 $ sabemos que la solución no está definida globalmente hasta $\displaystyle \infty  $. Por otro lado, si $\displaystyle \overline{x}\left(t_{0}\right) < 0 $ se tiene que 
\[ \left|\overline{x}\left(t\right)-\varphi\left(t; t_{0}, x_{0}\right)\right| = \frac{ \left|\overline{x}\left(t_{0}\right)-x_{0}\right|}{ \left|1-\overline{x}\left(t_{0}\right)\left(t-t_{0}\right)\right| \left|1-x_{0}\left(t-t_{0}\right)\right|} ,\]
de donde se sigue la estabilidad asintótica tomando $\displaystyle \delta \min \left\{ \epsilon , \left|\overline{x}\left(t_{0}\right)\right|\right\}  $ y $\displaystyle \eta = \left|\overline{x}\left(t_{0}\right)\right| > 0 $. 
\end{eg}
\begin{eg}
Consideremos la EDO no lineal $\displaystyle x '= x\left(1- x\right) $, cuyo flujo se puede obtener de forma explícita:
\[\varphi\left(t; t_{0}, x_{0}\right)=\frac{x_{0}}{x_{0} + \left(1-x_{0}\right)e^{t-t_{0}}} .\]
En este caso se puede comprobar que $\displaystyle \overline{y}\left(t\right)\equiv 0 $ es solución inestable mientras que $\displaystyle \overline{z}\left(t\right)\equiv 1 $ es asintóticamente estable.
\end{eg}

\begin{lema}[Reducción de estabilidad a la trivial]
Consideremos el sistema $\displaystyle x' = f\left(t,x\right) $ y una solución $\displaystyle \overline{x} : [\tau, \infty) \to \R^{n} $. Dada la función
\[g\left(t,y\right)=f\left(t,y + \overline{x}\left(t\right)\right)-f\left(t, \overline{x}\left(t\right)\right) ,\]
entonces
\begin{itemize}
\item $\displaystyle \overline{y}\left(t\right)\equiv 0 $ es solución de $\displaystyle y' = g\left(t,y\right) $. 
\item $\displaystyle \overline{x} $ es solución estable (resp. asintóticamente estable) de $\displaystyle x' = f\left(t,x\right) $ si y sólo si $\displaystyle \overline{y} $ es solución estable (resp. asintóticamente estable) de $\displaystyle y' = g\left(t,y\right) $. 
\end{itemize}
\end{lema}
\begin{proof}
En primer lugar, tenemos que 
\[g\left(t, 0\right)=f\left(t, \overline{x}\left(t\right)\right)-f\left(t, \overline{x}\left(t\right)\right)=0 = \overline{y}'\left(t\right) ,\]
por lo que $\displaystyle \overline{y}\left(t\right)\equiv 0 $ es solución de $\displaystyle y' = g\left(t,y\right) $, $\displaystyle \forall t \geq \tau $. Ahora, sea $\displaystyle \varphi\left(t; t_{0}, x_{0}\right) $ el flujo de $\displaystyle x' = f\left(t,x\right) $ y $\displaystyle \phi\left(t; t_{0}, x_{0}\right) $ el flujo de $\displaystyle y' = g\left(t,y\right) $. Entonces es sencillo comprobar que, 
\[\phi\left(t; t_{0}, y_{0}\right)=\varphi\left(t, t_{0}, y_{0} + \overline{x}\left(t_{0}\right)\right)-\overline{x}\left(t_{0}\right) .\]
Si cogemos $\displaystyle z\left(t\right):=\varphi\left(t, t_{0}, y_{0} + \overline{x}\left(t_{0}\right)\right)-\overline{x}\left(t\right)  $, tenemos que 
\[
\begin{split}
	z'\left(t\right)= &\frac{d}{dt}\left(\varphi\left(t, t_{0}, y_{0} + \overline{x}\left(t_{0}\right)\right)-\overline{x}\left(t\right) \right)=\frac{d\varphi}{dt}\left(t; t_{0}, y_{0}+\overline{x}\left(t_{0}\right)\right)-\overline{x}'\left(t\right)\\
	= & f\left(t; \varphi\left(t; t_{0}, y_{0} + \overline{x}\left(t_{0}\right)\right)\right)-f\left(t, \overline{x}\left(t\right)\right) 
	=  f\left(t, z\left(t\right)+\overline{x}\left(t\right)\right)-f\left(t, \overline{x}\left(t\right)\right) \\
	= &  g\left(t, z\left(t\right)\right).
\end{split}
\]
Además, 
\[z\left(t_{0}\right)=\varphi\left(t_{0}; t_{0}, y_{0} + \overline{x}\left(t_{0}\right)\right)-\overline{x}\left(t_{0}\right)=y_{0} + \overline{x}\left(t_{0}\right)-\overline{x}\left(t_{0}\right)=y_{0} .\]
Por unicidad de solución del PVI se tiene que $\displaystyle z\left(t\right)=\phi\left(t; t_{0}, y_{0}\right) $. Así, es sencillo ver que 
\[ \left|\phi\left(t; t_{0}, y_{0}\right)-\overline{y}\left(t\right)\right|= \left|\varphi\left(t; t_{0}, y_{0} + \overline{x}\left(t_{0}\right)\right)-\overline{x}\left(t\right)\right| ,\]
de donde se sigue inmediatamente la equivalencia de estabilidad.
\end{proof}
\begin{observation}
Gracias a este resultado, a partir de ahora podremos centrarnos en estudiar la estabilidad de la solución trivial $\displaystyle \overline{x}\equiv 0 $ en sistemas cuyo campo satisface $\displaystyle f\left(t,0\right)=0 $.
\end{observation}

\section{Estabilidad para sistemas lineales}
Consideremos el sistema 
\[x' = A\left(t\right)x + b\left(t\right) ,\]
con $\displaystyle A\in \mathcal{C}\left(I, \R^{n \times m}\right) $ y $\displaystyle b \in \mathcal{C}\left(I, \R^{n}\right) $, con $\displaystyle I $ un intervalo de $\displaystyle \R $. 
\begin{observation}
Con estas hipótesis de $\displaystyle A $ y $\displaystyle b $ podemos tomar
\[f\left(t,x\right)=A\left(t\right)x + b\left(t\right) ,\]
y $\displaystyle f $ cumple las hipótesis del teorema Picard Lindelöf local. En efecto, 
\[ \|f\left(t,x\right)-f\left(t,y\right)\| = \|A\left(t\right)\left(x-y\right)\| \leq \|A\left(t\right)\| \|x-y\| .\]
\end{observation}
\begin{prop}
Todas las soluciones de un sistema lineal $\displaystyle x' = A\left(t\right)x + b\left(t\right) $ tienen la misma estabilidad y además coincide con la de la solución $\displaystyle y \equiv 0 $ del sistema homogéneo $\displaystyle y' = A\left(t\right)y $. 
\end{prop}
\begin{proof}
Usando el lema anterior, dada una solución $\displaystyle \overline{x}\left(t\right) $ podemos construir 
\[
\begin{split}
	g\left(t,y\right)= & f\left(t,y+\overline{x}\left(t\right)\right)-f\left(t, \overline{x}\left(t\right)\right) \\
	= &  \left(A\left(t\right)\left(y + \overline{x}\left(t\right)+b\left(t\right)\right)\right)-\left(A\left(t\right)\overline{x}\left(t\right)+b\left(t\right)\right)=A\left(t\right)y.
\end{split}
\]
El sistema modificado que obtenemos es el homogéneo, por lo que la estabilidad de la solución $\displaystyle \overline{x}\left(t\right) $ equivale a la estabilidad de la solución trivial $\displaystyle \overline{y}\equiv 0 $ del sistema homogéneo. 
\end{proof}
\begin{observation}[Recordatorio]
El flujo del sistema $\displaystyle x' = A\left(t\right)x $ tiene la forma 
\[\varphi\left(t; t_{0}, x_{0}\right)=\Phi\left(t; t_{0}\right)x_{0} ,\]
donde $\displaystyle \Phi\left(t, t_{0}\right) $ es la \textbf{matriz fundamental}. Además, 
\[
\begin{cases}
\frac{d\Phi}{dt}\left(t; t_{0}\right)=A\left(t\right)\Phi\left(t; t_{0}\right) \\
\Phi\left(t_{0}; t_{0}\right)=I
\end{cases}
.\]
Por la propiedad del semigrupo tenemos que $\displaystyle \varphi\left(t; t_{0}, x_{0}\right)=\varphi\left(t; s, \varphi\left(s; t_{0}, x_{0}\right)\right)  $ por lo que
\[ \Phi\left(t; t_{0}\right)x_{0}=\Phi\left(t,s\right)\varphi\left(s; t_{0}, x_{0}\right)=\Phi\left(t,s\right)\Phi\left(s;t_{0}\right)x_{0}.\]
Como esto es cierto para todo $\displaystyle x_{0}\in \R^{n} $, debe ser que $\displaystyle \Phi\left(t; t_{0}\right)=\Phi\left(t; s\right)\Phi\left(s; t_{0}\right) $. En particular, si $\displaystyle t = t_{0} $,
\[I = \Phi\left(t_{0}; t_{0}\right)=\Phi\left(t_{0}, s\right)\Phi\left(s; t_{0}\right) .\]
\end{observation}

\begin{theorem}[Criterio de estabilidad para sistemas lineales homogéneos]
Un sistema lineal homogéneo $\displaystyle y' = A\left(t\right)y $, es estable si y sólo si todas sus soluciones son acotadas en $\displaystyle [t_{0}, \infty) $. 
\end{theorem}
\begin{proof}
Consideramos ambas implicaciones.
\begin{description}
\item[($\displaystyle \Rightarrow $)] Como el sistema es estable, tenemos que para cada $\displaystyle \epsilon > 0 $ y para cada $\displaystyle t_{0} \in \R $, existe $\displaystyle \delta > 0 $ tal que si $\displaystyle \|x_{0}\| < \delta  $ entonces $\displaystyle \|\varphi\left(t; t_{0}, x_{0}\right)\|< \epsilon  $. Estamos cogiendo $\displaystyle \overline{x}\equiv 0 $. Elegimos $\displaystyle \epsilon = 1 $, por tanto existe $\displaystyle \delta > 0 $ tal que si $\displaystyle \|x_{0}\| < \delta  $ entonces $\displaystyle \|\varphi\left(t; t_{0}, x_{0}\right)\| < 1 $. 
	Si $\displaystyle x_{0}\in \R^{n} $,
	\[
	\begin{split}
		\|\varphi\left(t; t_{0}, x_{0}\right)\| = & \|\Phi\left(t; t_{0}, x_{0}\right)\| = \left\|\Phi\left(t; t_{0}\right)\frac{x_{0}}{\|x_{0}\|}\frac{\delta }{2}\right\|\frac{2\|x_{0}\|}{\delta } \\
		= &  \left\|\varphi\left(t; t_{0}, \frac{x_{0}\delta }{2\|x_{0}\|}\right)\right\|\frac{2\|x_{0}\|}{\delta } < \frac{2\|x_{0}\|}{\delta }.
	\end{split}
	\]
	Así, la solución está acotada. 
\item[($\displaystyle \Leftarrow $)] Sabemos que $\displaystyle \|\varphi\left(t; t_{0}, x_{0}\right)\| $ está acotada. En particular, la matriz fundamental está acotada. Así, $\displaystyle \|\Phi\left(t, t_{0}\right)\| \leq C $, $\displaystyle \forall t \geq t_{0} $. Entonces, $\displaystyle \forall \epsilon > 0 $ cogemos $\displaystyle \delta = \frac{\epsilon }{C} $ y hacemos un proceso como el anterior. 
\end{description}
\end{proof}
\begin{theorem}
Un sistema lineal homogéneo es asintóticamente estable si y sólo si todas las soluciones tienden a cero cuando $\displaystyle t \to \infty  $. 
\end{theorem}
\begin{proof}
Consideremos ambas implicaciones.
\begin{description}
\item[($\displaystyle \Rightarrow $)] Si es asintóticamente estable, existe $\displaystyle \eta > 0 $ tal que si $\displaystyle \|x_{0}\| < \eta  $, entonces $\displaystyle \|\varphi\left(t; t_{0}, x_{0}\right)\| \to 0$ si $\displaystyle t \to \infty  $. 
	Cogemos $\displaystyle x_{0} \in \R $, tenemos que 
	\[ \|\varphi\left(t; t_{0}, x_{0}\right)\| = \|\Phi\left(t; t_{0}\right)x_{0}\| = \left\|\Phi\left(t, t_{0}\right)\frac{x_{0}}{\|x_{0}\|}\frac{\eta }{2}\right\| \frac{2\|x_{0}\|}{\eta} = \left\|\varphi\left(t; t_{0}, \frac{x_{0}}{\|x_{0}\|} \frac{\eta }{2}\right)\right\| \frac{2\|x_{0}\|}{\eta}\to 0.\]
\item[($\displaystyle \Leftarrow $)] Como todas las soluciones tienden a cero, en particular las columnas de la matriz fundamental tienden a 0. Es decir, $\displaystyle \Phi_{ij}\left(t, t_{0}\right)\to 0 $ si $\displaystyle t \to \infty  $. Así, $\displaystyle \forall i,j \in \left\{ 1, \ldots, n\right\}  $ y $\displaystyle \forall \epsilon > 0 $, existe $\displaystyle T$ tal que 
	\[ \left|\Phi_{ij}\left(t, t_{0}\right)\right|<\epsilon, \; \forall t > T .\]
En general, para cada $\displaystyle \epsilon > 0 $ podemos encontrar $\displaystyle T $ tal que 
\[ \left|\Phi_{ij}\left(t, t_{0}\right)\right|<\epsilon, \; \forall t > T, \; \forall i,j \in \left\{ 1, \ldots, n\right\}  .\]
Por tanto, 
\[ \|\Phi\|_{F}^{2} = \sum^{n}_{i,j = 1} \left|\Phi_{ij}\left(t, t_{0}\right)\right|^{2} < n^{2}\epsilon  ,\]
donde $\displaystyle \|\Phi\|_{F} $ es la norma de Frobenius de la matriz $\displaystyle \Phi $. Dado $\displaystyle x_{0} \in \R^{n} $, tenemos que 
\[ \|\varphi\left(t; t_{0}, x_{0}\right)\| = \|\Phi\left(t, t_{0}\right)x_{0}\| \leq \|\Phi\left(t; t_{0}\right)\| \|x_{0}\| \leq n^{2}\epsilon \|x_{0}\|, \; \forall t > T .\]
Por tanto, da igual cómo elijamos $\displaystyle \eta  $ que tenemos asegurada la convergencia a cero. 
\end{description}
\end{proof}
\begin{eg}[Problema 24, Hoja 2]
	Tenemos un sistema $\displaystyle x' = A\left(t\right)x $ con $\displaystyle A\left(t\right)=A\left(t + T\right) $ para cierto $\displaystyle T \in \R $. Sabemos que existe una solución $\displaystyle x\left(t\right)\not\equiv 0 $ y que es $\displaystyle T $-periódica. Queremos ver que no es asintóticamente estable. Si lo fuera, por lo que acabamos de ver, tendríamos que tendería a 0 cuando $\displaystyle t \to \infty $, que no sucede al ser $\displaystyle T $-periódica y no idénticamente nula.
\end{eg}
Supongamos ahora que tenemos un sistema de la forma
\[x' = Ax, \; A \in \R^{n \times n} ,\]
con coeficientes constantes. Tendremos que $\displaystyle \Phi\left(t, t_{0}\right)=e^{A\left(t-t_{0}\right)} $. Recordemos que dada una matriz $\displaystyle M \in \R^{n \times n} $ 
\[e^{M} = Id + \sum^{\infty}_{k = 1}\frac{M^{k}}{k!}, \; e^{0}= Id .\]
\begin{prop}[Propiedades de la exponencial de matrices]
\begin{itemize}
\item Si $\displaystyle M $ y $\displaystyle N $ son matrices tales que $\displaystyle M N = N M $, entonces $\displaystyle e^{M + N}=e^{M}e^{N} $. 
\item Si $\displaystyle M = JNJ^{-1} $, entonces $\displaystyle e^{M}=Je^{N}J^{-1} $. 
\item Si $\displaystyle M = diag\left(M_{1}, \ldots, M_{l}\right) $ es diagonal por bloques, entonces $\displaystyle e^{M}=diag\left(e^{M_{1}}, \ldots, e^{M_{l}}\right) $. 
\item Si tenemos una matriz 
	\[D_{r}\left(\lambda \right)=\begin{pmatrix} \lambda & \cdots & 0 \\ 1 & \cdots & 0 \\ \vdots & & \vdots \\ 0 & \cdots & 1\end{pmatrix}\in \R^{r \times r} .\]
Tendremos que 
\[e^{D_{r}\left(\lambda \right)t}=e^{\lambda t}  .\]
% copiar lo del año pasado de EDOS
\end{itemize}
\end{prop}
Si tenemos una matriz $\displaystyle M $, para calcular la exponencial de $\displaystyle M $ calcularíamos primero la forma de Jordan de $\displaystyle M $. A continuación, podemos utilizar \textbf{(2)} y \textbf{(3)} para hacer la exponencial en cada bloque, que tendrá la forma de $\displaystyle e^{D_{r}\left(\lambda \right)t} $ que viene en \textbf{(4)}. \\
\\
Si $\displaystyle r $ es la diferencia de multiplicidades más uno, 
\[\left(e^{tA}\right)_{ij} = \sum_{\lambda \in \sigma \left(A\right)}e^{\lambda t}P_{r, \lambda, i,j}\left(t\right)=\sum_{\lambda \in \sigma \left(A\right)}e^{\Re\left(\lambda \right)t}\left(A \cos \left(\Imagen \lambda t\right) + B\sin \Imagen\lambda t\right) .\]

\begin{prop}
	Dada una matriz $\displaystyle A\in \R^{n \times n} $ y $\displaystyle \alpha_{\max }= \max_{\lambda \in \sigma \left(A\right)} \left\{ \Re\left(\lambda \right)\right\}  $. Entonces, para todo $\displaystyle \gamma > \alpha_{\max } $, existe $\displaystyle C \geq 0 $ tal que 
	\[\|e^{tA}\|_{F}\leq Ce^{\gamma t} .\]
	Si además para todo $\displaystyle \lambda\in \sigma\left(A\right) $ tal que $\displaystyle \Re\left(\lambda \right)=\alpha_{\max } $ se tiene que $\displaystyle m_{g}\left(\lambda \right)=m_{a}\left(\lambda \right) $ entonces
	\[\|e^{tA}\|_{F}\leq C_{\alpha_{\max }}e^{\alpha_{\max }t} .\]	
\end{prop}
\begin{proof}
Si $\displaystyle \gamma > \alpha_{\max } $ podemos coger $\displaystyle \epsilon = \gamma - \alpha_{\max } $ y tenemos que existe $\displaystyle C > 0 $ tal que 
\[ \left|P_{r, \lambda, i,j}\left(t\right)\right|\leq Ce^{\epsilon t} .\]
Entonces, 
\[ \left|\left(e^{tA}\right)_{ij}\right|\leq \sum_{\lambda \in \sigma\left(A\right)}e^{\Re\left(\lambda \right)t}Ce^{\epsilon t} \leq \sum_{\lambda \in \sigma\left(A\right)}e^{\alpha_{\max }t + \epsilon t}C = C_{\gamma }^{i,j}e^{\left(\alpha_{\max }+\epsilon \right)t}=C^{i,j}_{\gamma }e^{\gamma t} .\]
Entonces, 
\[\|e^{tA}\|^{2}_{F}\leq e^{2\gamma t}\sum_{i,j}\left(C_{\gamma }^{ij}\right)^{2} .\]
Para el otro caso, 
\[\tilde{\alpha}_{\max }=\max_{\lambda \in \sigma \left(A\right), \Re\left(\lambda \right)<\alpha_{\max }} \left\{ \Re\left(\lambda \right)\right\} < \alpha_{\max } .\]
Así, tomamos $\displaystyle \epsilon = \alpha_{\max }-\tilde{\alpha}_{\max } > 0 $. Entonces, 
\[ \left|P_{r,\lambda, i,j}\left(t\right)\right|\leq C_{r,\lambda,i,j,\epsilon }e^{\epsilon t}\quad \text{si} \quad \Re\left(\lambda \right)< \alpha_{\max } .\]
En particular, $\displaystyle \left|P_{r,x,i,j}\left(t\right)\right|\leq C_{r,\lambda,i,j,\epsilon } $ si $\displaystyle \Re\left(\lambda \right)=\alpha_{\max } $ por lo que el polinomio es constante. Entonces, 
\[
\begin{split}
	\left|\left(e^{tA}\right)_{ij}\right| \leq & \sum_{\lambda \in \sigma\left(A\right), \Re\left(\lambda \right)=\max }e^{\alpha_{\max }t}C + \sum_{ \lambda \in \sigma\left(A\right), \Re\left(\lambda \right)< \alpha_{\max }}e^{\Re\left(\lambda \right)t}Ce^{\epsilon t} 
	\leq  Ce^{\alpha_{\max }t}.
\end{split}
\]
\end{proof}
\begin{theorem}
Dado $\displaystyle A \in \R^{n \times n} $, entonces $\displaystyle x' = Ax $ es
\begin{itemize}
\item asintóticamente estable si y sólo si $\displaystyle \Re\left(\lambda \right)<0 $ para todo $\displaystyle \lambda \in \sigma\left(A\right) $. 
\item estable si y sólo si $\displaystyle \Re\left(\lambda \right)\leq 0 $ para todo $\displaystyle \lambda \in \sigma\left(A\right) $ y además $\displaystyle m_{g}\left(\lambda \right)=m_{a}\left(\lambda \right) $ para todo $\displaystyle \lambda \in \sigma \left(A\right) $ con $\displaystyle \Re\left(\lambda \right)=0 $. 
\end{itemize}
\end{theorem}
\begin{proof}
\begin{itemize}
\item $\displaystyle \Re\left(\lambda \right)<0 $ para todo $\displaystyle \lambda \in \sigma\left(A\right) $, entonces 
	\[\alpha_{\max }=\max _{\lambda \in \sigma\left(A\right)} \left\{ \Re\left(\lambda \right)\right\} <0 \Rightarrow \exists \gamma \in \left(\alpha_{\max }, 0\right) .\]
Por la propiedad anterior, $\displaystyle \|e^{tA}\|_{F}\leq C_{\gamma }e^{\gamma t} $ por lo que 
\[ \left|\varphi\left(t; t_{0}, x_{0}\right)\right| = \left|e^{\left(t-t_{0}\right)A}x_{0}\right|\leq \|e^{\left(t-t_{0}\right)A}\|_{F} \left|x_{0}\right| \leq C_{\gamma } \left|x_{0}\right|e^{\gamma\left(t-t_{0}\right)} \xrightarrow{\gamma < 0, t \to \infty }0 .\]
Así, es asintóticamente estable. 
\item Si $\displaystyle \Re\left(\lambda \right)\leq 0 $ y $\displaystyle m_{g}\left(\lambda \right) = m_{a}\left(\lambda \right) $, $\displaystyle \forall \lambda \in \sigma\left(A\right) $ tal que $\displaystyle \Re\left(\lambda \right)=0 $, entonces $\displaystyle \alpha_{\max }=0 $ y $\displaystyle m_{g}\left(\lambda \right)=m_{a}\left(\lambda \right) $ para todo $\displaystyle \lambda  $ con $\displaystyle \Re\left(\lambda \right)=\alpha_{\max } $. 
Así, por la proposición anterior tenemos que 
\[\|e^{tA}\| \leq C_{\alpha_{\max }}e^{\alpha_{\max }t} = C_{0} .\]
Entonces, 
\[ \left|\varphi\left(t; t_{0}, x_{0}\right)\right|\leq \|e^{\left(t-t_{0}\right)A}\|_{F} \left|x_{0}\right| \leq C_{0} \left|x_{0}\right| .\]
Como todas las sucesiones están acotadas tenemos que el sistema es estable. 
\end{itemize}
\end{proof}

\begin{eg}[Problema 25]
Sea $\displaystyle x ' = Ax $ un sistema estable. Tenemos que $\displaystyle c : [\tau, \infty) \to \R^{n \times n} $ es una función continua tal que 
\[\int^{\infty}_{\beta } \sum^{n}_{i,j = 1} \left|c_{ij}\left(s\right)\right| \; d s < \infty .\]
Queremos ver que el sistema $\displaystyle x'=\left(A + c\left(t\right)\right)x $ es estable. La condición que nos dan es equivalente a que 
\[ \int^{\infty}_{\beta } \|c\left(s\right)\|_{F} \; d s < \infty .\]
Como $\displaystyle x' = Ax $ es estable existe $\displaystyle C > 0 $ tal que $\displaystyle \|e^{At}\|_{F}\leq C $, $\displaystyle \forall t \geq \tau $. Si tenemos la solución del problema $\displaystyle x' = \left(A + c\left(t\right)\right)x $ que denotamos por $\displaystyle \varphi\left(t; t_{0}, x_{0}\right) $, por la proposición
\[\varphi\left(t; t_{0}, x_{0}\right) = e^{A\left(t-t_{0}\right)}x_{0} + \int^{t}_{t_{0}} e^{A\left(t-s\right)}c\left(s\right)\varphi\left(s; t_{0}, x_{0}\right) \; d s .\]
Por tanto, $\displaystyle \forall t \in [t_{0}, \beta_{+ }) $,
\[
\begin{split}
	\left|\varphi\left(t; t_{0}, x_{0}\right)\right| \leq & \|e^{A\left(t-t_{0}\right)}\| \left|x_{0}\right| + \int^{t}_{t_{0}} \|e^{A\left(t-s\right)}\| \|c\left(s\right)\| \left|\varphi\left(s; t_{0}, x_{0}\right)\right| \; d s  \\
	\leq & C \left|x_{0}\right| +C \int^{t}_{t_{0}} \|c\left(s\right)\| \left|\varphi\left(s; t_{0}, x_{0}\right)\right| \; d s .
\end{split}
\]
Aplicando la desigualdad de Grönwall, 
\[ \left|\varphi\left(t; t_{0}, x_{0}\right)\right| \leq C \left|x_{0}\right|\exp\left(C\int^{t}_{t_{0}} \|c\left(s\right)\| \; d s\right)\leq C \left|x_{0}\right|\exp\left(C\int^{\infty }_{\tau} \|c\left(s\right)\|  \; d s\right) ,\]
por tanto, tenemos que todas las soluciones están acotadas y por ser un sistema lineal debe ser que el sistema es estable.
