\chapter{Teoría local de Cauchy y aplicaciones}
\section{Preliminares}
\begin{definition}[Curva]
	Una \textbf{curva} es una aplicación continua $\displaystyle \gamma : \left[a,b\right] \to \R^{n} $. 
\end{definition}
\begin{notation}
	Para evitar problemas asumiremos que $\displaystyle \gamma  $ es $\displaystyle \mathcal{C}^{1} $ y que $\displaystyle \gamma'\neq 0 $ en $\displaystyle \left[a,b\right]  $. 
\end{notation}
Vamos a establecer la siguiente relación de equivalencia entre curvas:
\[\gamma_{1} \sim \gamma_{2} \iff \exists\varphi:\left[c,d\right] \to \left[a,b\right] , \; \varphi' > 0, \; \gamma_{2}=\gamma_{1}\circ \varphi .\]
Podemos observar que $\displaystyle \varphi $ es un difeomorfismo que preserva la orientación de la curva. Así, podemos considerar el conjunto cociente correspondiente obteniendo el \textbf{conjunto de curvas orientadas} $\displaystyle \mathcal{C}^{1} $. Esto nos permite identificar cada curva con su conjunto imagen $\displaystyle \Gamma = \left[\gamma \right]  $. 
\begin{notation}
	Si la orientación es la opuesta escribimos $\displaystyle \Gamma^{-}=\left[\gamma^{-}\right]  $. En particular, dada $\displaystyle \gamma : \left[a,b\right] \to \R^{n} $ tendremos que $\displaystyle \gamma^{-}\left(t\right)=\gamma\left(b+a-t\right) $. 
\end{notation}
\begin{definition}[Curva $\displaystyle \mathcal{C}^1 $ a trozos]
	Decimos que una curva $\displaystyle \gamma : \left[a,b\right] \to \R^{n} $ es $\displaystyle \mathcal{C}^{1} $ \textbf{a trozos} si existe una partición de $\displaystyle \left[a,b\right]  $, $\displaystyle a_{0}= a < a_{1} < \cdots < a_{m} = b $ tal que la restricción de $\displaystyle \gamma  $ a cada subintervalo $\displaystyle \left[a_{j}, a_{j+1}\right]  $ para $\displaystyle j = 0, \ldots, m - 1 $ define una curva parametrizada de clase $\displaystyle \mathcal{C}^{1} $.
\end{definition}

Como vimos en cálculo integral, todo lo anterior se puede generalizar a curvas que sean $\displaystyle \mathcal{C}^{1} $ a trozos. 
\begin{definition}[Longitud de una curva]
	Dada una curva $\displaystyle \gamma : \left[a,b\right] \to \R^{n} $ de clase $\displaystyle \mathcal{C}^{1} $ a trozos, se define la \textbf{longitud} de $\displaystyle \gamma  $ como
	\[\ell\left(\gamma \right)=\int^{b}_{a} \|\gamma'\left(t\right)\| \; dt .\]
\end{definition}
Es sencillo comprobar que si $\displaystyle \gamma  $ y $\displaystyle \sigma  $ son curvas equivalentes, entonces $\displaystyle \ell\left(\gamma \right)=\ell\left(\sigma \right) $, también $\displaystyle \ell\left(\gamma \right)=\ell\left(\gamma^{-} \right) $. De esta forma, para cada curva orientada $\displaystyle \Gamma  $ de clase $\displaystyle \mathcal{C}^{1} $ a trozos puede definirse $\displaystyle \ell\left(\Gamma \right)=\ell\left(\gamma \right) $ con $\displaystyle \Gamma = \left[\gamma \right]  $. 
Además, $\displaystyle \ell\left(\Gamma \right)=\ell\left(\Gamma^{-}\right) $. \\

Una operación que usamos con frecuencia es la \textbf{concatenación de curvas}. Si $\displaystyle \Gamma_{1}, \ldots, \Gamma_{m} $ son curvas de clase $\displaystyle \mathcal{C}^{1} $ a trozos en $\displaystyle \R^{n} $ tales que el punto final de $\displaystyle \Gamma_{j} $ coincide con el punto inicial de $\displaystyle \Gamma_{j + 1} $, se puede definir una curva $\displaystyle \Gamma $ que recorre de manera consecutiva $\displaystyle \Gamma_{1} , \ldots, \Gamma_{m} $ que denotaremos por $\displaystyle \Gamma = \Gamma_{1} \cup \cdots \cup \Gamma_{m} $. 
Para encontrar una parametrización de $\displaystyle \Gamma  $ basta con tomar parametrizaciones $\displaystyle \gamma_{1}, \ldots, \gamma_{m} $ de $\displaystyle \Gamma_{1}, \ldots, \Gamma_{m} $ definidas en intervalos $\displaystyle \left[a_{j}, b_{j}\right]  $ para $\displaystyle j \in \left\{ 1, \ldots, m\right\}  $ de manera que $\displaystyle b_{j}=a_{j+1} $ y que $\displaystyle \gamma_{j}\left(b_{j}\right)=\gamma_{j+1}\left(a_{j+1}\right) $. Así, definimos
\[\gamma : [a_{1}, b_{m}] \to \R^{n} : t \to \gamma_{j}\left(t\right), \; t \in \left[a_{j}, b_{j}\right]  .\]
Claramente $\displaystyle \Gamma $ será la clase de las curvas equivalentes a esta $\displaystyle \gamma  $. 
\begin{definition}
Sea $\displaystyle \Gamma $ una curva orientada de clase $\displaystyle \mathcal{C}^{1} $ a trozos en $\displaystyle \R^{n} $. 
\begin{itemize}
	\item Diremos que $\displaystyle \Gamma $ es una \textbf{curva cerrada} si admite una parametrización $\displaystyle \gamma : \left[a,b\right] \to \R^{n} $ tal que $\displaystyle \gamma\left(a\right)=\gamma\left(b\right) $.
	\item Diremos que $\displaystyle \Gamma $ es una \textbf{curva simple} si admite una parametrización $\displaystyle \gamma : \left[a,b\right] \to \R^{n} $ que es inyectiva.
	\item Diremos que $\displaystyle \Gamma $ es una \textbf{curva cerrada simple} si admite una parametrización $\displaystyle \gamma : \left[a,b\right] \to \R^{n} $ tal que $\displaystyle \gamma\left(a\right)=\gamma\left(b\right) $ y la restricción de $\displaystyle \gamma  $ a $\displaystyle [a,b) $ es inyectiva.
\end{itemize}
\end{definition}
\begin{observation}
Para las curvas paramétricas cerradas $\displaystyle \mathcal{C}^{1} $ a trozos se pueden considerar reparametrizaciones $\displaystyle \gamma : \mathbb{S}^{1} \to \R^{n} $ para evitar problemas. Usando este punto de vista
\[\gamma \sim \sigma \iff \exists \varphi : \mathbb{S}^{1} \to \mathbb{S}^{1} ,\]
homeomorfismo $\displaystyle \mathcal{C}^{1} $ a trozos que preserva orientación tal que $\displaystyle \sigma =\gamma\circ\varphi $. 
\end{observation}
\section{Integración de funciones con valores en $\displaystyle \R^n $}
Sea $\displaystyle \varphi : A \subset \R^{d}\to \R^{n} $, podemos escribir $\displaystyle \varphi = \left(\varphi_{1}, \ldots, \varphi_{n}\right) $ y definir
\[\int _{A} \varphi \; d x : = \left(\int _{A}\varphi_{1} \; dx, \ldots, \int _{A}\varphi_{n} \; dx\right),\]
siempre que las integrales $\displaystyle \int _{A}\varphi_{i} \; dx $ estén bien definidas $\displaystyle \forall i \in \left\{ 1, \ldots, n\right\}  $. 
\begin{prop}[Propiedades de la integral]
Sean $\displaystyle \varphi, \psi : A\subset \R^{d} \to \R^{n} $ integrables y $\displaystyle \alpha, \beta \in \R $. Entonces, 
\begin{enumerate}
\item $\displaystyle \int _{A}\left(\alpha\varphi + \beta\varphi\right) \; dx = \alpha\int _{A}\varphi \; dx + \beta\int _{A}\psi \; dx $. 
\item $\displaystyle \left\|\int _{A}\varphi \; dx\right\| \leq \int _{A}\|\varphi\| \; dx$. 
\item \textbf{Teorema fundamental del cálculo}. Para toda $\displaystyle g : \left[a,b\right] \to \R^{n} $ de clase $\displaystyle \mathcal{C}^{1}\left(\left[a,b\right] \right) $ se tiene que 
	\[g\left(t\right)=g\left(a\right)+\int^{t}_{a} g'\left(x\right) \; dx .\]
\item Si $\displaystyle A = A_{1} \cup A_{2} $ con $\displaystyle m\left(A_{1} \cap A_{2}\right)=0 $, entonces $\displaystyle \int _{A}\varphi \; dx = \int _{A_{1}}\varphi \; dx + \int _{A_{2}}\varphi \; dx $. 
\end{enumerate}
\end{prop}
\begin{proof}
Las propiedades \textbf{(1)}, \textbf{(3)} y \textbf{(4)} se deducen fácilmente a partir de sus resultados correspondientes para funciones con valores en $\displaystyle \R $. Para demostrar \textbf{(2)} denotamos 
\[x = \left(x_{1}, \ldots, x_{n}\right)=\left(\int _{A}\varphi_{1} , \ldots, \int _{A}\varphi_{n} \right)=\int _{A}\varphi .\]
Podemos suponer que $\displaystyle x \neq 0 $. Usando la linealidad de la integral de funciones con valores en $\displaystyle \R $ y la desigualdad Cauchy-Schwarz en $\displaystyle \R^{n} $,
\[\|x\|^{2}=\left\langle x, x \right\rangle  =\sum^{n}_{j= 1}x_{j}\int _{A}\varphi_{j}\left(t\right) \; dt = \int _{A}\sum^{n}_{j = 1}x_{j}\varphi_{j}\left(t\right) \; dt \leq \int _{A}\|x\| \|\varphi\| \; dt = \|x\|\int _{A}\|\varphi\|,\]
deducimos \textbf{(2)} al dividir por $\displaystyle \|x\| $. 
\end{proof}

\section{Integración de funciones complejas en $\displaystyle \C $}
Al identificar $\displaystyle \R^{2} $ con $\displaystyle \C $ podemos ver que dada $\displaystyle \varphi : \left[a,b\right] \to \C $ con $\displaystyle \varphi\left(t\right)=x\left(t\right) + iy\left(t\right) $ se define de manera natural la \textbf{integral compleja} como
	\[\int^{b}_{a} \varphi\left(t\right) \; dt := \int^{b}_{a} x\left(t\right) \; dt + i\int^{b}_{a} y\left(t\right) \; dt .\]
	Podemos observar que por la propiedad \textbf{(2)} anterior tendremos que 
\[ \left|\int^{b}_{a} \varphi\left(t\right) \; dt\right| \leq \int^{b}_{a} \left|\varphi\left(t\right)\right| \; dt .\]
\begin{definition}[Integral compleja sobre una curva]
	Sea $\displaystyle f : \Omega\subset \C \to \C $ continua y $\displaystyle \gamma : \left[a,b\right] \to \Omega  $ una curva parametrizada de clase $\displaystyle \mathcal{C}^{1} $ a trozos. Se define la \textbf{integral de $\displaystyle f $ sobre $\displaystyle \gamma  $} como
	\[\int _{\gamma }f : = \int^{b}_{a} f\left(\gamma\left(t\right)\right)\gamma'\left(t\right) \; dt .\]
\end{definition}
Si escribimos $\displaystyle f = u + iv $ y $\displaystyle \gamma\left(t\right)=x\left(t\right)+iy\left(t\right) $, obtenemos una expresión equivalente para la integral,
\[
\begin{split}
	\int _{\gamma }f = & \int^{b}_{a} \left(u\left(x\left(t\right), y\left(t\right)\right)x'\left(t\right)-v\left(x\left(t\right), y\left(t\right)\right)y'\left(t\right)\right) \; dt 
\\
			  + & i \int^{b}_{a} \left(u\left(x\left(t\right), y\left(t\right)\right)y'\left(t\right) + v\left(x\left(t\right), y\left(t\right)\right)x'\left(t\right)\right) \; dt..
\end{split}
\]

\begin{observation}
El producto que aparece en la integral anterior \textbf{no es el producto escalar}, es el producto usual definido en $\displaystyle \C $. Esto tiene varias consecuencias:
\begin{enumerate}
\item El valor de $\displaystyle \int _{\gamma }f $ es en general un número complejo por lo que no coincide con el valor de la integral del campo vectorial $\displaystyle f $ a lo largo de $\displaystyle \gamma  $ en $\displaystyle \R^{2} $ (sería un valor real).
\item \textbf{No es cierto en general} que $\displaystyle \left|\int _{\gamma }f \right|\leq \int _{\gamma } \left|f\right| $. Esto no tendría sentido puesto que, como hemos dicho antes, el término de la derecha es en general un número complejo. Ni siquiera tiene por qué cumplirse aún suponiendo que ambos términos son reales, como se verá en un ejemplo a continuación. 
\end{enumerate}
\end{observation}
\begin{eg}
	Cogemos $\displaystyle f\left(z\right)=\frac{1}{z} $ en $\displaystyle \mathbb{S}^{1} $. Observamos que si cogemos la parametrización $\displaystyle \gamma\left(t\right)=e^{it} $ con $\displaystyle t \in \left[0, 2\pi \right]  $ tendremos que 
	\[\int _{\gamma }\frac{1}{z} \; dz = \int^{2\pi }_{0} \frac{1}{e^{it}}ie^{it} \; dt = 2\pi i \Rightarrow  \left|\int _{\gamma }\frac{1}{z} \right|=2\pi.\]
Sin embargo, 
\[\int_{\gamma} \left|f\right| = \int _{\gamma }1 =\int^{2\pi }_{0} 1 \cdot ie^{it} =0 .\]
La desigualdad $\displaystyle \left|\int _{\gamma }f \right|\leq \int _{\gamma } \left|f\right| $ no se cumple puesto que $\displaystyle 2\pi > 0 $. 
\end{eg}

\begin{prop}[Propiedades de la integral compleja]
Sean $\displaystyle f, g : \Omega \subset \C \to \C $ continuas, $\displaystyle \alpha, \beta \in \C $ y $\displaystyle \gamma : \left[a,b\right] \to \Omega  $ curva parametrizada $\displaystyle \mathcal{C}^{1} $ a trozos. Entonces, 
	\begin{enumerate}
	\item $\displaystyle \int _{\gamma }\alpha f + \beta g = \alpha \int_{\gamma } f +\beta \int_{\gamma} g  $.
	\item $\displaystyle \int _{\gamma^{-} }f =-\int _{\gamma }f $. 
	\item $\displaystyle \left|\int _{\gamma }f \right|\leq \sup_{z \in \gamma\left(\left[a,b\right] \right)} \left|f\left(z\right)\right| \ell\left(\gamma \right) $.
	\end{enumerate}
\end{prop}
\begin{proof}
Las propiedades \textbf{(1)} y \textbf{(2)} se vieron en cálculo integral. Demostramos \textbf{(3)}, 
\[
\begin{split}
	\left|\int _{\gamma }f \right|= & \left|\int^{b}_{a} f\left(\gamma\left(t\right)\right)\gamma'\left(t\right) \; dt\right|\leq \int^{b}_{a} \left|f\left(\gamma\left(t\right)\right)\right| \left|\gamma'\left(t\right)\right| \; dt \\
	\leq & \sup_{z \in \gamma\left(\left[a,b\right] \right)} \left|f\left(z\right)\right| \int^{b}_{a} \left|\gamma'\left(t\right)\right| \; dt = \sup_{z \in \gamma\left(\left[a,b\right] \right)} \left|f\left(z\right)\right| \ell\left(\gamma \right).
\end{split}
\]
\end{proof}
\begin{prop}
	Sean $\displaystyle \gamma : \left[a,b\right] \to \Omega  $ y $\displaystyle \sigma : \left[c,d\right] \to \Omega  $ curvas parametrizadas de clase $\displaystyle \mathcal{C}^{1} $ a trozos y supongamos que son equivalentes. Entonces, para toda función continua $\displaystyle f : \Omega \subset \C \to \C $ se tiene que 
	\[\int _{\gamma }f =\int _{\sigma  }f  .\]
\end{prop}
\begin{proof}
	Haremos la demostración suponiendo que $\displaystyle \gamma  $ y $\displaystyle \sigma  $ son $\displaystyle \mathcal{C}^{1} $ puesto que la extensión al caso general es evidente. Supongamos pues que $\displaystyle \phi : \left[c,d\right] \to \left[a,b\right]  $ es biyectiva de clase $\displaystyle \mathcal{C}^{1} $ con $\displaystyle \phi'\left(t\right)>0 $ con $\displaystyle \sigma\left(s\right)=\gamma\left(\phi\left(s\right)\right) $. Aplicando el teorema de cambio de variable tenemos que 
	\[\int _{\gamma }f =\int^{b}_{a} f\left(\gamma\left(t\right)\right)\gamma'\left(t\right) \; dt = \int^{d}_{c} f\left(\gamma\left(\phi\left(s\right)\right)\right)\gamma'\left(\phi\left(s\right)\right)\phi'\left(s\right) \; d s = \int^{d}_{c} f\left(\sigma\left(s\right)\right) \sigma'\left(s\right) \; d s = \int _{\sigma }f .\]
\end{proof}

Debido a este hecho, para cada $\displaystyle \Gamma $ curva orientada de clase $\displaystyle \mathcal{C}^{1} $ a trozos contenida en $\displaystyle \C $ tendremos que podemos definir
	\[\int _{\Gamma}f := \int _{\gamma }f ,\]
con $\displaystyle \gamma  $ parametrización de $\displaystyle \Gamma  $, para cualquier función continua $\displaystyle f : \Gamma \to \C $. Las propiedades de la proposición anterior tienen análogos obvios para $\displaystyle \int _{\Gamma }f $. 
Asímismo, dada una concatenación de curvas $\displaystyle \Gamma = \bigcup^{m}_{j = 1}\Gamma_{j} $, es sencillo ver que
\[\int _{\Gamma}f = \sum^{m}_{j =1}\int _{\Gamma_{j}}f .\]
En el caso particular de $\displaystyle  = 2 $ y de que $\displaystyle \Gamma = \Gamma_{1} \cup\Gamma_{2} $ sea cerrada tenemos que $\displaystyle \int _{\Gamma }f  $ no depende del punto inicial o final. En efecto,
\[\int _{\Gamma }f =\int _{\Gamma_{1}}f +\int _{\Gamma_{2}}f =  \int _{\Gamma_{2}}f +\int _{\Gamma_{1}}f  .\]
Entonces, si $\displaystyle \gamma, \omega : \mathbb{S}^{1} \to \C $ son equivalentes 
\[\int _{\gamma }f =\int _{\sigma } f .\]
En efecto, si $\displaystyle z = \gamma\left(1\right) $ y $\displaystyle \omega = \sigma\left(1\right) $ son distintos, 
\[\int _{\gamma }f =\int _{\Gamma_{1}}f +\int _{\Gamma_{2}}f =\int _{\Gamma_{2}}f +\int _{\Gamma_{1}}f =\int _{\sigma }f ,\]
donde $\displaystyle \Gamma_{1} $ denota el subarco de $\displaystyle \gamma\left(\mathbb{S}^{1}\right)=\sigma\left(\mathbb{S}^{1}\right) $ que comienza en $\displaystyle z $ y acaba en $\displaystyle w $ y se recorre en el mismo sentido que lo hacen $\displaystyle \gamma  $ y $\displaystyle \sigma  $; y $\displaystyle \Gamma_{2} $ es el subarco que comienza en $\displaystyle w $ y acaba en $\displaystyle z $ y se recorre en el mismo sentido que $\displaystyle \gamma  $ y $\displaystyle \sigma  $. 
\begin{definition}[Función primitiva]
Dado $\displaystyle \Omega \subset \C $ abierto y $\displaystyle f : \Omega \to \C $, decimos que $\displaystyle F : \Omega \to \C $ con $\displaystyle F \in \mathcal{H}\left(\Omega \right) $ es una \textbf{primitiva} de $\displaystyle f$ si $\displaystyle F'\left(z\right)=f\left(z\right) $, $\displaystyle \forall z \in \Omega  $. 
\end{definition}
\begin{observation}
Si $\displaystyle \Omega \subset \C $ es abierto conexo y $\displaystyle F_{1} $ y $\displaystyle F_{2} $ son primitivas de $\displaystyle f $, debe ser que $\displaystyle F_{1} $ y $\displaystyle F_{2} $ son iguales salvo quizás una constante.  
\end{observation}
\begin{theorem}
Sea $\displaystyle \Omega\subset \C $ abierto, $\displaystyle f \in \mathcal{C}\left(\Omega \right) $ y $\displaystyle F $ una primitiva de $\displaystyle f $ en $\displaystyle \Omega  $. Sea $\displaystyle \Gamma $ una curva de clase $\displaystyle \mathcal{C}^{1} $ a trozos que empieza en $\displaystyle \omega_{1} $ y termina en $\displaystyle \omega_{2} $. Entonces,
\[\int _{\Gamma }f =F\left(\omega_{2}\right)-F\left(\omega_{1}\right) .\]
\end{theorem}
\begin{proof}
	Supongamos que $\displaystyle \Gamma $ es $\displaystyle \mathcal{C}^{1} $. Sea $\displaystyle \gamma : \left[a,b\right] \to \Omega $ una parametrización de $\displaystyle \Gamma  $. Ahora, aplicando la regla de la cadena compleja 
	\[
	\begin{split}
	\int _{\gamma }f =\int^{b}_{a} f\left(\gamma\left(t\right)\right)\gamma'\left(t\right) \; dt= \int^{b}_{a} \frac{d}{dt}F\left(\gamma\left(t\right)\right) \; dt = F\left(\gamma\left(b\right)\right)-F\left(\gamma\left(a\right)\right)=F\left(\omega_{2}\right)-F\left(\omega_{1}\right) .
	\end{split}
	\]
En el caso de que $\displaystyle \Gamma $ sea $\displaystyle \mathcal{C}^{1} $ a trozos tendremos que 
\[\int _{\Gamma}f =\sum^{n}_{j = 1}\int _{\Gamma }f =\sum^{n}_{j = 1}\left(F\left(\omega_{2}^{j}\right)-F\left(\omega^{j}_{1}\right)\right) = F\left(\omega_{2}^{n}\right)-F\left(\omega_{1}^{1}\right) .\]
\end{proof}
\begin{colorary}
Si $\displaystyle \Gamma $ es una curva cerrada $\displaystyle \mathcal{C}^{1} $ a trozos y $\displaystyle f : \Omega\subset \C \to \C $, $\displaystyle f \in \mathcal{C}\left(\Omega \right) $ que admite primitiva en $\displaystyle \Omega  $, entonces $\displaystyle \int _{\Gamma }f =0 $. 
\end{colorary}
\begin{eg}
	La función $\displaystyle f\left(z\right)=\frac{1}{z} $ no admite primitiva en $\displaystyle \C\backslash \left\{ 0\right\}  $. En efecto, por el corolario anterior tenemos que 
	\[\int _{ \left\{ \left|z\right| = 1\right\} }\frac{1}{z} \; dz = \int^{2\pi }_{0} \frac{1}{e^{it}}ie^{it} \; dt =2\pi i .\]
\end{eg}
\begin{theorem}[Derivación bajo el signo de la integral]
Sean $\displaystyle \Omega\subset \C $ abierto, $\displaystyle \Gamma $ una curva orientada de clase $\displaystyle \mathcal{C}^{1} $ a trozos (no pedimos que esté en $\displaystyle \Omega  $). Sea $\displaystyle \varphi : \Gamma \times \Omega \to \C $ continua tal que $\displaystyle \forall \xi \in \Gamma  $ la aplicación $\displaystyle z \to \varphi\left(\xi, z\right)\in \mathcal{H}\left(\Omega \right) $ y tal que $\displaystyle \left(\xi, z\right)\to \frac{\partial \varphi}{\partial z}\left(\xi, z\right) $ es continua en $\displaystyle \Gamma \times \Omega  $.
Entonces
\[g\left(z\right):= \int _{\Gamma}\varphi\left(\xi, z\right) \; d\xi ,\]
es holomorfa en $\displaystyle z \in \Omega  $ y 
\[g'\left(z\right)=\int _{\Gamma}\frac{\partial\varphi}{\partial z}\left(\xi, z\right) \; d\xi .\]
\end{theorem}
\begin{proof}
	Sea $\displaystyle z_{0} \in \Omega  $ y $\displaystyle r > 0 $ tal que $\displaystyle \overline{B}\left(z_{0}, r\right)\subset \Omega  $. Para $\displaystyle \left|h\right|<r $, considerando el segmento $\displaystyle S = \left[z_{0}, z_{0} + h\right] \subset \Omega  $ parametrizado por $\displaystyle S_{h}\left(t\right)=z_{0} + th $, $\displaystyle t \in \left[0,1\right]  $, y aplicando el teorema anterior tenemos que 
	\[\varphi\left(\xi, z_{0} + h\right)-\varphi\left(\xi, z_{0}\right)=\int _{S_{h}}\frac{\partial\varphi}{\partial z}\left(\xi, z\right) \; dz = \int^{1}_{0} \frac{\partial \varphi}{\partial z}\left(\xi, S_{h}\right)S'_{h}\left(t\right) \; dt = \int^{1}_{0} \frac{\partial\varphi}{\partial z}\left(\xi,z_{0}+th\right)h \; dt \]
	para cada $\displaystyle \xi \in \Gamma $. Suponiendo que $\displaystyle \gamma : \left[a,b\right] \to \Omega  $ es una parametrización de $\displaystyle \Gamma $ tenemos que
	\[
	\begin{split}
		&\lim_{h \to 0}\frac{g\left(z_{0} + h\right)-g\left(z_{0}\right)}{h} =  \lim_{h \to 0}\frac{1}{h}\int _{\Gamma}\left(\varphi\left(\xi, z_{0} + h\right)-\varphi\left(\xi, z_{0}\right)\right) \; d\xi \\
		= &  \lim_{h \to 0}\frac{1}{h}\int _{\Gamma }\left(\int^{1}_{0} \frac{\partial\varphi}{\partial z}\left(\xi, z_{0} + sh\right)h \; d s\right) \; d\xi 
		=  \lim_{h \to 0}\int^{b}_{a} \left(\int^{1}_{0} \frac{\partial \varphi}{\partial z}\left(\gamma\left(t\right), z_{0} + sh\right) \; d s\right)\gamma'\left(t\right) \; dt \\
		= &  \lim_{h \to 0}\int _{\left[0,1\right] \times\left[a,b\right] }\frac{\partial \varphi}{\partial z}\left(\gamma\left(t\right), z_{0} + sh\right)\gamma'\left(t\right) \; d s \; dt  = \int _{\left[0,1\right] \times\left[a,b\right] }\lim_{h \to 0}\frac{\partial \varphi}{\partial z}\left(\gamma \left(t\right), z_{0} + sh\right)\gamma'\left(t\right) \; d s dt\\
		= & \int _{\left[0,1\right] \times\left[a,b\right] }\frac{\partial \varphi}{\partial z}\left(\gamma\left(t\right), z_{0}\right)\gamma'\left(t\right) \; d s \; dt = \int^{b}_{a} \frac{\partial \varphi}{\partial z}\left(\gamma\left(t\right), z_{0}\right)\gamma'\left(t\right) \; dt = \int _{\Gamma}\frac{\partial \varphi}{\partial z}\left(\xi, z\right) \; d\xi.
	\end{split}
	\]
En la cuarta igualdad hemos usado el teorema de Fubini y en la quinta el intercambio de la integral y el límite se justifica porque al ser $\displaystyle \frac{\partial \varphi}{\partial z} $ continua en el compacto $\displaystyle \Gamma \times \overline{B}\left(z_{0}, r\right) $ es uniformemente continua en este conjunto y por tanto
\[\lim_{h \to 0}\frac{\partial \varphi}{\partial z}\left(\gamma\left(t\right), z_{0} + sh\right)\gamma'\left(t\right)=\frac{\partial \varphi}{\partial z}\left(\gamma\left(t\right), z_{0}\right)\gamma'\left(t\right)\]
uniformemente en $\displaystyle \left(s,t\right)\in \left[0,1\right] \times\left[a,b\right]  $. 
\end{proof}
\begin{colorary}
Sean $\displaystyle z_{0} \in \C $ y $\displaystyle r > 0 $. Para todo $\displaystyle z \in D\left(z_{0}, r\right) $ es
\[\int _{ \left\{ \left|z-z_{0}\right|=r\right\} }\frac{1}{\xi-z} \; d\xi = 2\pi i ,\]
con orientación positiva. 
\end{colorary}
\begin{proof}
Tomamos 
\[\varphi: \partial D\left(z_{0}, r\right)\times \C \backslash \partial D\left(z_{0}, r\right)\to \C : \left(\xi, z \right)\to \frac{1}{\xi - z} .\]
Por construcción el denominador nunca se anula. En el dominio, la función $\displaystyle \varphi $ es continua y su derivada parcial es $\displaystyle \frac{\partial \varphi}{\partial z}=\frac{1}{\left(\xi-z\right)^{2}} $ que también es continua en el dominio. 
Así, la función
\[f\left(z\right):= \int _{ \left\{ \left|z-z_{0}\right|=r\right\} }\frac{1}{\xi-z} \; d\xi ,\]
está bien definida y por el teorema anterior es holomorfa en $\displaystyle D\left(z_{0}, r\right) $ con derivada
\[f'\left(z\right)=\int _{ \left\{ \left|z-z_{0}\right|=r\right\} }\frac{\partial \varphi}{\partial z} \; d\xi = \int _{ \left\{ \left|z-z_{0}\right|=r\right\} }\frac{1}{ \left(\xi-z\right)^{2}} \; d\xi .\]
Esta integral es nula para cada $\displaystyle z \in D\left(z_{0}, r\right) $ puesto que es la integral en una curva cerrada de una función que es continua y tiene primitiva $\displaystyle -\frac{1}{\xi-z} $. Se sigue que $\displaystyle f $ es constante en $\displaystyle D\left(z_{0}, r\right) $, por ser este un abierto conexo. Por ende, $\displaystyle \forall z \in D\left(z_{0}, r\right) $,
\[f\left(z\right)=f\left(z_{0}\right)=\int _{ \left\{ \left|z-z_{0}\right|=r\right\} }\frac{1}{\xi-z_{0}} \; d\xi =\int^{2\pi }_{0} \frac{rie^{it}}{re^{it}} \; dt = 2\pi i .\]
Hemos usdo la parametrización $\displaystyle \gamma\left(t\right)=z_{0} + re^{it} $ con $\displaystyle t \in \left[0,2\pi \right]  $. 
\end{proof}
\section{Teorema de Cauchy-Goursat}
\begin{theorem}[Teorema de Goursat]
	Sean $\displaystyle \Omega \subset \C $ un abierto, $\displaystyle T \subset \Omega  $ el borde de un triángulo con su interior contenido en $\displaystyle \Omega  $. Sea $\displaystyle f \in \mathcal{H}\left(\Omega \backslash \left\{ p\right\} \right) $ con $\displaystyle p \in \Omega  $. Entonces, $\displaystyle \int _{T}f =0 $.
\end{theorem}
\begin{proof}
	Sea $\displaystyle T = T_{0} $ el triángulo dado, con vértices $\displaystyle a,b,c \in \C $ y sea $\displaystyle \mathcal{T} = \mathcal{T}_{0} $ la unión de $\displaystyle T $ con la región interior a $\displaystyle T $ con borden $\displaystyle \partial \mathcal{T}=T $. Podemos suponer que $\displaystyle T $ está orientado positivamente y que sus lados se recorren en este orden: $\displaystyle \left[a,b\right]  $, $\displaystyle \left[b,c\right]  $, $\displaystyle \left[c,a\right]  $. Podemos distinguir varios casos:
	\begin{description}
		\item[Caso 1.] Supongamos que $\displaystyle p\not\in \mathcal{T} $. Uniendo los puntos medios de los lados $\displaystyle \left[a,b\right]  $, $\displaystyle \left[b,c\right]  $ y $\displaystyle \left[c,a\right]  $ obtenemos un triángulo que denotamos $\displaystyle T^{4}_{0} $ que divide $\displaystyle \mathcal{T}^{0}_{0} $ en cuatro subtriángulos sólidos $\displaystyle \mathcal{T}^{j}_{0} $ con bordes $\displaystyle T^{j}_{0} $ con $\displaystyle j = 1, \ldots, 4 $, cuyos perímetros son la mitad del perímetro del triángulo original $\displaystyle T $. 
% añadir imagen
Elijamos en los triángulos $\displaystyle T^{j}_{0} $ con $\displaystyle 1 \leq j \leq 3 $ las orientaciones inducidas por la orientación de $\displaystyle T $, y orientemos cada lado de $\displaystyle T^{4}_{0} $ de forma opuesta a la del mismo lado considerado como lado de $\displaystyle T^{j}_{0} $ para algún $\displaystyle j = 1, 2, 3 $ (equivalentemente, orientamos todos los triángulos $\displaystyle T^{j}_{0} $ de forma positiva). Entonces, tenemos que 
\[\int _{T}f =\sum^{4}_{j=1}\int _{T^{j}_{0}}f  ,\]
puesto que las integrales de la suma de la derecha se cancelan mutuamente en cada lado de $\displaystyle T^{4}_{0} $. Debe existir algún $\displaystyle j \in \left\{ 1, \ldots, 4\right\}  $ tal que 
\[ \left|\int _{T_{0}^{j}}f \right|\geq \frac{1}{4} \left|\int _{T_{0}}f \right| .\]
En caso contrario tendríamos que 
\[ \left|\int _{T_{0}}f \right|= \left|\sum^{4}_{j=1}\int _{T_{0}^{j}}f \right|<\sum^{4}_{j=1}\frac{1}{4} \left|\int _{T_{0}}f \right|= \left|\int _{T_{0}}f \right| .\]
Al $\displaystyle T^{j}_{0} $ que hemos elegido lo llamamos $\displaystyle T_{1} $, denotando por $\displaystyle \mathcal{T}_{1} $ la región interior a $\displaystyle T_{1} $ junto con el propio $\displaystyle T_{1} $. Repetimos el argumento anterior cambiando $\displaystyle T_{0} $ por $\displaystyle T_{1} $. Dividimos $\displaystyle \mathcal{T}_{1} $ en cuatro subtriángulos sólidos $\displaystyle \mathcal{T}^{j}_{1} $ con bordes $\displaystyle T^{j}_{1} $ con $\displaystyle j = 1, \ldots, 4 $, cuyos perímetros serán un cuerto del perímetro del triángulo original $\displaystyle T $. 
Elegimos uno de los cuatro triángulos $\displaystyle T^{j}_{1} $ para el que se tenga 
\[ \left|\int _{T^{j}_{1}}f \right|\geq \frac{1}{4} \left|\int _{T_{1}}f \right| .\]
De esta forma, 
\[ \left|\int _{T^{j}_{1}}f \right|\geq \frac{1}{4^{2}} \left|\int _{T}f \right| .\]
Reiterando este proceso obtenemos por inducción una sucesión $\displaystyle \left\{ \mathcal{T}_{j}\right\} _{j \in \N} $ de triángulos sólidos con bordes $\displaystyle T_{j} $ tales que 
\begin{enumerate}
\item $\displaystyle \mathcal{T}_{j}\subset\mathcal{T}_{j-1} $, $\displaystyle \forall j \in \N $ con $\displaystyle \mathcal{T}_{0}=\mathcal{T} $. 
\item $\displaystyle \left|\int _{T_{j}}f \right|\geq \frac{1}{4^{j}} \left|\int _{T}f \right| $. 
\item $\displaystyle \ell\left(T_{j}\right)=\frac{1}{2^{j}}\ell\left(T\right) $.  
\end{enumerate}
En particular, como la longitud de un triángulo es mayor o igual que su diámetro, se tiene que $\displaystyle \left\{ \mathcal{T}_{j}\right\} _{j \in \N} $ es una familia de compactos en $\displaystyle \C \equiv \R^{2} $ cuyos diámetros tienden a cero, por lo que existe un único $\displaystyle z_{0} \in \C $ tal que 
\[z_{0}\in \bigcap_{j \in \N}\mathcal{T}_{j} .\]
Puesto que estamos suponiendo que $\displaystyle p \not\in \mathcal{T} $, la función $\displaystyle f $ es holomorfa en $\displaystyle z_{0} $ y por tanto, dado cualquier $\displaystyle \epsilon > 0 $ existe $\displaystyle \delta > 0 $ tal que si $\displaystyle \left|z-z_{0}\right|\leq \delta  $, entonces
\[ \left|f\left(z\right)-f\left(z_{0}\right)-f'\left(z_{0}\right)\left(z-z_{0}\right)\right|\leq \epsilon \left|z-z_{0}\right| .\]
Por otro lado, como $\displaystyle \lim_{j \to \infty}\ell\left(T_{j}\right)=0 $, existe $\displaystyle j_{0} \in \N $ tal que si $\displaystyle j \geq j_{0} $ entonces
\[diam\left(T_{j}\right)\leq \ell\left(T_{j}\right)<\delta ,\]
luego $\displaystyle \left|z-z_{0}\right|\leq \delta  $ para todo $\displaystyle z \in T_{j} $, de donde deducimos que para $\displaystyle j \geq j_{0} $, 
\[
\begin{split}
	\left|\int _{T_{j}}f \right|= & \left|\int _{T_{j}}f\left(z\right)-f\left(z_{0}\right)-f'\left(z_{0}\right)\left(z-z_{0}\right) \; dz\right|\\
	\leq& \epsilon \sup_{z\in T_{j}} \left|z-z_{0}\right| \ell\left(T_{j}\right) \leq \epsilon \ell\left(T_{j}\right)^{2}=\frac{\epsilon }{4^{j}}\ell\left(T\right)^{2},
\end{split}
\]
donde en la primera igualdad hemos usado que 
\[\int _{T_{j}}f\left(z_{0}\right)+f'\left(z_{0}\right)\left(z-z_{0}\right) \; dz = 0 ,\]
por ser la integral en una curva cerrada de una función que tiene primitiva. En la segunda desigualdad hemos vuelto a usar que el diámetro de un triángulo es menor o igual que su perímetro, y en la última igualdad hemos utilizado la tercera propiedad de la sucesión de triángulos. Combinando la desigualdad anterior con la propiedad \textbf{(2)} de la sucesión de triángulos se obtiene que 
\[ \frac{1}{4^{j}} \left|\int _{T}f \right|\leq \left|\int _{T_{j}}f \right|\leq \frac{\epsilon }{4^{j}}\ell\left(T\right)^{2} \Rightarrow \left|\int _{T}f \right|\leq \epsilon \ell\left(T\right)^{2} .\]
Como es cierto para todo $\displaystyle \epsilon > 0 $ debe ser que $\displaystyle \int _{T}f =0 $. 
\item[Caso 2.] Supongamos que $\displaystyle p \in \mathcal{T} $. Podemos distinguir tres casos. 
	\begin{description}
	\item[Caso 2.1.] Si $\displaystyle p $ es un vértice de $\displaystyle T $ podemos suponer sin pérdida de generalidad que $\displaystyle p = a $. Como $\displaystyle f $ es continua en $\displaystyle T $ compacto, existe $\displaystyle M > 0 $ tal que $\displaystyle \left|f\left(z\right)\right|\leq M $, $\displaystyle \forall z \in T $. 
		Dado $\displaystyle \epsilon > 0 $, consideremos los puntos $\displaystyle x \in \left[a,b\right]  $ e $\displaystyle y \in \left[c,a\right]  $ tales que la longitud del triángulo $\displaystyle T_{1} $ de vértices $\displaystyle a,x,y $ sea menor que $\displaystyle \frac{\epsilon }{M} $. 
		Denotamos por $\displaystyle T_{2} $ el triángulo de vértices $\displaystyle x,b,y $ y por $\displaystyle \mathcal{T}_{3} $ el de vértices $\displaystyle b,c,y $ con las orientaciones inducidas por la de $\displaystyle T $. Por lo demostrado en el \textbf{caso 1} tenemos que 
		\[\int _{T_{2}}f =\int _{T_{3}}f =0 .\]
		Por tanto, 
		\[ \left|\int _{T}f \right|= \left|\sum^{3}_{j=1}\int _{T_{j}}f \right|= \left|\int _{T_{1}}f \right|\leq M \ell\left(T_{1}\right) \leq \epsilon  .\]
		Como esto es cierto para todo $\displaystyle \epsilon > 0 $ debe ser que $\displaystyle \int _{T}f =0 $. 
	\item[Caso 2.2.] Supongamos que $\displaystyle p $ está en un lado de $\displaystyle T $. Digamos por ejemplo que $\displaystyle p \in \left[a,b\right]  $. Sea $\displaystyle T_{1} $ el triángulo de vértices $\displaystyle a,p,c $ y $\displaystyle T_{2} $ el triángulo de vértices $\displaystyle p,b,c $. Usando el caso anterior tenemos que 
		\[\int _{T}f =\int _{T_{1}}f +\int _{T_{2}}f = 0 + 0 = 0 .\]
	\item[Caso 2.3.] Supongamos que $\displaystyle p $ está en el interior de $\displaystyle \mathcal{T} $. Definimos $\displaystyle T_{1} $ como el triángulo de vértices $\displaystyle a,p,c $; $\displaystyle T_{2} $ el de vértices $\displaystyle a,b,p $; y $\displaystyle T_{3} $ el de $\displaystyle b,c,p $. 
		Usando el \textbf{caso 2.1} tenemos que 
		\[\int _{T}f =\int _{T_{1}}f +\int _{T_{2}}f +\int _{T_{3}}f =0 + 0 + 0 = 0 .\]
	\end{description}
	\end{description}
\end{proof}

\begin{theorem}[Existencia de primitivas en abiertos convexos]
	Sean $\displaystyle \Omega\subset \C $ abierto convexo, $\displaystyle p \in \Omega  $ y $\displaystyle f : \Omega \to \C $ continua tal que $\displaystyle f $ es holomorfa en $\displaystyle \Omega \backslash \left\{ p\right\}  $. Entonces existe $\displaystyle F : \Omega \to \C $ holomorfa tal que $\displaystyle F'=f $.
\end{theorem}
\begin{proof}
Fijamos $\displaystyle z_{0}\in \Omega  $ y definimos la función
\[F : \Omega \to \C : z \to \int _{\left[z_{0}, z\right] }f  ,\]
donde $\displaystyle \left[z_{0}, z\right]  $ es el segmento que une $\displaystyle z_{0} $ con $\displaystyle z $, orientado en la dirección $\displaystyle z-z_{0} $. Además, este conjunto está contenido en $\displaystyle \Omega  $ por ser este convexo. Fijamos ahora $\displaystyle z \in \Omega  $ y tomamos $\displaystyle r > 0 $ tal que $\displaystyle D\left(z,r\right)\subset \Omega  $. Para cada $\displaystyle h \in \C $ con $\displaystyle \left|h\right|<r $ tenemos que, gracias a la convexidad de $\displaystyle \Omega  $, 
el triángulo $\displaystyle T_{z,h} $ de vértices $\displaystyle z_{0}, z, z+h $, así como su región interio, está contenido en $\displaystyle \Omega  $. Por el teorema de Goursat tenemos que 
\[\int _{T_{z,h}}f =0 .\]
Equivalentemente, 
\[\int _{\left[z_{0}, z\right] }f +\int _{\left[z, z+h\right] }f +\int _{\left[z+h, z_{0}\right] }f =0 \Rightarrow F\left(z + h\right)-F\left(z\right)=\int _{\left[z, z+h\right] }f\left(\xi\right) \; d\xi .\]
Dado que $\displaystyle f\left(z\right)h=\int _{\left[z, z + h\right] }f\left(z\right) \; d\xi $, tenemos que 
\[
\begin{split}
	\left|F\left(z + h\right)-F\left(z\right)-f\left(z\right)h\right| = & \left|\int _{\left[z, z+h\right] }f\left(\xi\right)-f\left(z\right) \; d\xi\right| \\
	\leq & \sup_{\xi\in\left[z, z + h\right] } \left|f\left(\xi\right)-f\left(z\right)\right|\ell\left(\left[z , z + h\right] \right)\\
	= & \sup_{\xi\in \left[z, z+ h\right] } \left|f\left(\xi\right)-f\left(z\right)\right| \left|h\right| = o\left(h\right).
\end{split}
\]
Como $\displaystyle f $ es continua en $\displaystyle z $ tenemos que $\displaystyle \lim_{h \to 0}\sup_{\xi\in [z, z + h]} \left|f\left(\xi\right)-f\left(z\right)\right|=0 $. Esto prueba que existe $\displaystyle F'\left(z\right) $ igual a $\displaystyle f\left(z\right) $. 
\end{proof}
\begin{observation}
	La demostración anterior funciona igual si se cambia la hipótesis de que $\displaystyle f $ sea holomorfa en $\displaystyle \Omega \backslash \left\{ p\right\}  $ por la de que $\displaystyle \int _{T}f =0 $ para cada triángulo $\displaystyle T $ contenido en $\displaystyle \Omega  $. 
\end{observation}
\begin{theorem}[Teorema integral de Cauchy, versión 1]
	Sean $\displaystyle \Omega \subset \C $ abierto convexo, $\displaystyle p \in \Omega  $ y $\displaystyle f : \Omega \to \C $ continua tal que $\displaystyle f $ es holomorfa en $\displaystyle \Omega \backslash \left\{ p\right\}  $. Entonces, para toda curva cerrada $\displaystyle \Gamma $ en $\displaystyle \Omega  $ se tiene que 
	\[\int _{\Gamma }f =0 .\]
\end{theorem}
\begin{proof}
Por el teorema anterior, existe $\displaystyle F : \Omega \to \C $ holomorfa tal que $\displaystyle F'=f $ en $\displaystyle \Omega  $. Entonces, al ser $\displaystyle \Gamma $ una curva cerrada tendremos que $\displaystyle \int _{\Gamma}f =0 $.
\end{proof}
\section{Fórmula integral de Cauchy}
\begin{theorem}[Fórmula integral de Cauchy]
Sean $\displaystyle \Omega\subset \C $ un abierto y $\displaystyle f : \Omega \to \C $ holomorfa. Entonces, para cada disco cerrado $\displaystyle D $ contenido en $\displaystyle \Omega  $ se tiene que 
\[f\left(z\right)=\frac{1}{2\pi i}\int _{\partial D}\frac{f\left(\xi\right)}{\xi-z} \; d\xi ,\]
donde $\displaystyle \partial D $ está orientado positivamente, para todo $\displaystyle z \in \Int\left(D\right) $. 
\end{theorem}
\begin{proof}
Cambiando $\displaystyle \Omega  $ por un disco abierto ligeramente mayor que $\displaystyle D $ contenido en $\displaystyle \Omega  $, podemos suponer sin pérdida de generalidad que $\displaystyle \Omega  $ es convexo. Fijamos $\displaystyle z \in \Int\left(D\right) $ y definimos
\[g : \Omega \to \C : w \to 
\begin{cases}
\frac{f\left(w\right)-f\left(z\right)}{w-z}, \; \text{si} \; w \neq z \\
f'\left(z\right), \; \text{si} \; w = z
\end{cases}
.\]
Es sencillo ver que $\displaystyle g $ es continua en $\displaystyle \Omega  $ y holomorfa en $\displaystyle \Omega\backslash \left\{ z\right\}  $. Por el teorema anterior tenemos que 
\[\int _{\partial D}g\left(w\right) \; dw = 0 \Rightarrow \int _{\partial D}\frac{f\left(w\right)-f\left(z\right)}{w-z} \; dw = 0 .\]
De esta forma, 
\[0 = \int _{\partial D}\frac{f\left(w\right)}{w-z} \; dw - f\left(z\right)\int _{\partial D}\frac{1}{w-z} \; dw = \int _{\partial D}\frac{f\left(w\right)}{w-z} \; dw -f\left(z\right)2\pi i.\]
Despejando obtenemos la igualdad deseada.
\end{proof}
\begin{colorary}[Propiedad del valor medio]
Si $\displaystyle f $ es holomorfa en un entorno abierto de $\displaystyle \overline{D}\left(z_{0}, r\right) $ entonces
\[ f\left(z_{0}\right)=\frac{1}{2\pi }\int^{2\pi }_{0} f\left(z_{0} + re^{i\theta }\right) \; d\theta  .\]
\end{colorary}
\begin{proof}
	Por el teorema anterior, tenemos que 
	\[f\left(z_{0}\right)=\frac{1}{2\pi i}\int _{\partial D}\frac{f\left(\xi\right)}{\xi - z_{0}} \; d\xi .\]
	Cogiendo la parametrización $\displaystyle \gamma\left(\theta \right)=z_{0} + re^{i\theta } $ para $\displaystyle \theta \in \left[0, 2\pi \right]  $, nos queda
	\[
		f\left(z_{0}\right)=  \frac{1}{2\pi i}\int _{\partial D} \frac{f\left(\xi\right)}{\xi-z_{0}}\; d\xi  = \frac{1}{2\pi i}\int^{2\pi }_{0} \frac{f\left(z_{0} + re^{i\theta }\right)}{z_{0}+re^{i\theta }-z_{0}}rie^{i\theta } \; d\theta = \frac{1}{2\pi }\int^{2\pi }_{0} f\left(z_{0}+re^{i\theta }\right) \; d\theta.
	\]
\end{proof}
\begin{colorary}[Propiedad del valor medio de las funciones armónicas]
Si $\displaystyle u $ es armónica en un entorno abierto de $\displaystyle \overline{D}\left(z_{0}, r\right) $ entonces
\[u\left(z_{0}\right)=\frac{1}{2\pi }\int^{2\pi }_{0} u\left(z_{0}+re^{i\theta }\right) \; d\theta  .\]
\end{colorary}
\begin{theorem}[Fórmulas integrales de Cauchy para las derivadas]
Sean $\displaystyle \Omega \subset \C $ un abierto y $\displaystyle f : \Omega \to \C $ holomorfa. Entonces su derivada $\displaystyle f' $ también es holomorfa y en particular continua. De hecho, $\displaystyle f $ tiene derivadas de todos los órdenes y si $\displaystyle D $ es un disco cerrado contenido en $\displaystyle \Omega  $, la derivada $\displaystyle n $-ésima de $\displaystyle f $ satisface
\[f^{\left(n\right)}\left(z\right)=\frac{n!}{2\pi i}\int _{\partial D}\frac{f\left(\xi\right)}{\left(\xi-z\right)^{n+1}} \; d\xi ,\; \forall z \in \Int\left(D\right), \; \forall n \in \N \cup \left\{ 0\right\}. \]
\end{theorem}
\begin{proof}
Lo demostramos por inducción. 
\begin{itemize}
\item Si $\displaystyle n = 0 $, el enunciado es cierto puesto que se reduce a la fórmula integral de Cauchy.
\item Supongamos que la fórmula es válida para $\displaystyle n $, entonces aplicando el Teorema de derivación bajo el signo de la integral obtenemos 
	\[
	\begin{split}
		f^{\left(n+1\right)}\left(z\right)=&\frac{d}{dz}f^{\left(n\right)}\left(z\right) =\frac{d}{dz}\left(\frac{n!}{2\pi i}\int _{\partial D}\frac{f\left(\xi\right)}{\left(\xi-z\right)^{n+1}} \; d\xi\right)=\frac{n!}{2\pi i}\int _{\partial D}\frac{d}{dz}\frac{f\left(\xi\right)}{\left(\xi-z\right)^{n+1}} \; d\xi \\
		= & \frac{n!}{2\pi i}\int _{\partial D}\frac{\left(n+1\right)f\left(\xi\right)}{\left(\xi-z\right)^{n+2}} \; d\xi=\frac{\left(n+1\right)!}{2\pi i}\int _{\partial D}\frac{f\left(\xi\right)}{\left(\xi-z\right)^{n+2}} \; d\xi.
	\end{split}
	\]
\end{itemize}
Así, por el principio de inducción el enunciado y la fórmula son válidos para cualquier $\displaystyle n \in \N \cup \left\{ 0\right\}  $.
\end{proof}
\begin{colorary}[Desigualdades de Cauchy]
Si $\displaystyle f $ es holomorfa en un abierto de $\displaystyle \C $ que contiene un disco cerrado $\displaystyle D = \overline{D}\left(z_{0}, R\right) $, entonces
\[ \left|f^{\left(n\right)}\left(z_{0}\right)\right| \leq \frac{n!}{R^{n}}\sup_{z \in \partial D} \left|f\left(z\right)\right| , \; \forall n \in \N\]
\end{colorary}
\begin{proof}
Aplicando el teorema anterior,
\[
\begin{split}
	\left|f^{\left(n\right)}\left(z_{0}\right)\right| = & \left|\frac{n!}{2\pi i}\int _{\partial D}\frac{f\left(\xi\right)}{\left(\xi-z_{0}\right)^{n+1}} \; d\xi\right|=\frac{n!}{2\pi } \left|\int^{2\pi }_{0} \frac{f\left(z_{0}+Re^{it}\right)Rie^{it}}{\left(z_{0}+Re^{it}-z_{0}\right)^{n+1}} \; dt\right| \\
	\leq & \frac{n!}{2\pi }\int^{2\pi }_{0}  \left|\frac{f\left(z_{0} + Re^{it}\right)Rie^{it}}{\left(Re^{it}\right)^{n+1}}\right| \; dt = \frac{n!}{2\pi }\int^{2\pi }_{0} \frac{ \left|f\left(z_{0} + Re^{it}\right)\right|R}{R^{n+1}} \; dt 
	\leq \frac{n!}{R^{n}}\sup_{z \in \partial D} \left|f\left(z\right)\right|.
\end{split}
\]
\end{proof}
\begin{observation}
Se deduce inmediatamente de este corolario, aplicando el criterio de la raíz, que la serie
\[\sum^{\infty}_{n = 0}\frac{f^{\left(n\right)}\left(z_{0}\right)}{n!}\left(z-z_{0}\right)^{n} ,\]
tiene radio de convergencia mayor o igual que el radio de cualquier disco cerrado centrado en $\displaystyle z_{0} $. 
\end{observation}
\begin{theorem}[Analiticidad de las funciones holomorfas]
Sean $\displaystyle \Omega \subset \C $ abierto y $\displaystyle f : \Omega \to \C $ holomorfa. Entonces, para cada disco cerrado $\displaystyle D = \overline{D}\left(z_{0}, R\right) $ contenido en $\displaystyle \Omega  $, la función $\displaystyle f $ tiene una expansión en serie de potencias centrada en $\displaystyle z_{0} $,
\[f\left(z\right)=\sum^{\infty}_{n = 0}\frac{f^{\left(n\right)}\left(z_{0}\right)}{n!}\left(z-z_{0}\right)^{n} ,\]
que converge uniformemente a $\displaystyle f\left(z\right) $ en $\displaystyle z \in D $. 
\end{theorem}
\begin{proof}
Fijamos $\displaystyle z \in \Int\left(D\right) $. Para cada $\displaystyle \xi \in \partial D $ se tiene que 
\[ \left|\frac{z -z_{0}}{\xi -z_{0}}\right| = \frac{ \left|z-z_{0}\right|}{R} < 1 ,\]
por lo que podemos usar el criterio $\displaystyle M $ de Weierstrass obteniendo, 
\[\sum^{\infty}_{n = 0}\left(\frac{z-z_{0}}{\xi-z_{0}}\right)^{n}=\frac{1}{1-\left(\frac{z-z_{0}}{\xi-z_{0}}\right)} ,\]
converge uniformemente en $\displaystyle \xi \in \partial D $. Esto implica que 
\[\frac{1}{\xi -z}=\frac{1}{\xi-z_{0}-\left(z-z_{0}\right)} = \frac{1}{\xi-z_{0}}\frac{1}{1-\left(\frac{z-z_{0}}{\xi-z_{0}}\right)}=\frac{1}{\xi-z_{0}}\sum^{\infty}_{n = 0}\left(\frac{z-z_{0}}{\xi-z_{0}}\right)^{n}=\sum^{\infty }_{n=0}\frac{\left(z-z_{0}\right)^{n}}{\left(\xi-z_{0}\right)^{n+1}} ,\]
uniformemente en $\displaystyle \xi \in \partial D $. Por tanto, usando las integrales de Cauchy obtenemos 
\[
\begin{split}
	f\left(z\right)= & \frac{1}{2\pi i}\int _{\partial D}\frac{f\left(\xi\right)}{\xi-z} \; d\xi = \frac{1}{2\pi i}\int _{\partial D}f\left(\xi\right)\left(\sum^{\infty}_{n=0}\frac{\left(z-z_{0}\right)^{n}}{\left(\xi-z_{0}\right)^{n+1}}\right) \; d\xi 
	= \sum^{\infty}_{n=0}\left(\int _{\partial D}\frac{f\left(\xi\right)\left(z-z_{0}\right)^{n}}{\left(\xi-z_{0}\right)^{n+1}} \; d\xi\right) \\
	= & \sum^{\infty}_{n = 0}\left(\int _{\partial D}\frac{f\left(\xi\right)}{\left(\xi-z_{0}\right)^{n+1}} \; d\xi\right)\left(z-z_{0}\right)^{n}=\sum^{\infty}_{n=0}\frac{f^{\left(n\right)}\left(z_{0}\right)}{n!}\left(z-z_{0}\right)^{n},
\end{split}
\]
donde el intercambio de suma infinita e integral se justifica gracias a la convergencia uniforme de la suma en $\displaystyle \xi \in \partial D $. Esto prueba que la serie 
\[\sum^{\infty}_{n=0}\frac{f^{\left(n\right)}\left(z_{0}\right)}{n!}\left(z-z_{0}\right)^{n} ,\]
converge a $\displaystyle f\left(z\right) $ para cada $\displaystyle z \in \Int\left(D\right) $. Como el mismo argumento vale cambiando $\displaystyle \overline{D}\left(z_{0}, R\right) $ por $\displaystyle \overline{D}\left(z_{0}, R + \epsilon \right) $ para algún $\displaystyle \epsilon > 0 $ suficientemente pequeño tal que $\displaystyle \overline{D}\left(z_{0}, R + \epsilon \right) $ siga estando contenido en $\displaystyle \Omega  $, es obvio que la serie también converge a $\displaystyle f\left(z\right) $ para $\displaystyle z \in \partial D $. 
Finalmente, para dicho $\displaystyle \epsilon  $, de las desigualdades de Cauchy se deduce que 
\[ \left|\frac{f^{\left(n\right)}\left(z_{0}\right)}{n!}\right|^{\frac{1}{n}}\leq \frac{C^{\frac{1}{n}}}{\left(R+\epsilon \right)} ,\]
donde $\displaystyle C = \sup_{z \in \partial D\left(z_{0}, R + \epsilon \right)} \left|f\left(z\right)\right| $ puede suponerse positivo, luego
\[\frac{1}{\lim_{n \to \infty}\sup \left|\frac{f^{\left(n\right)}\left(z_{0}\right)}{n!}\right|^{\frac{1}{n}}}\geq \lim_{n \to \infty }\frac{R + \epsilon }{C^{\frac{1}{n}}} = R + \epsilon  .\]
Por la fórmula de Hadamard se deduce que la serie de potencias 
\[\sum^{\infty}_{n = 0}\frac{f^{\left(n\right)}\left(z_{0}\right)}{n!}\left(z-z_{0}\right)^{n} ,\]
tiene radio de convergencia mayor o igual que $\displaystyle R + \epsilon  $, y en particular converge uniformemente en el disco cerrrado $\displaystyle \overline{D}\left(z_{0}, R\right) $. 
\end{proof}
\begin{colorary}
El radio de convergencia de la serie de potencias centrada en $\displaystyle z_{0} $, 
\[\sum^{\infty}_{ n=0}\frac{f^{\left(n\right)}\left(z_{0}\right)}{n!}\left(z-z_{0}\right)^{n} ,\]
de una función holomorfa $\displaystyle f $ en un abierto $\displaystyle \Omega \subset \C $ es mayor o igual que 
\[\sup \left\{ r > 0 \; : \; D\left(z_{0}, r\right)\subset \Omega \right\} ,\]
es decir, mayor o igual que el radio del mayor disco abierto de centro $\displaystyle z_{0} $ que esté contenido en $\displaystyle \Omega  $.
\end{colorary}
\begin{observation}
Si $\displaystyle f : \C \to \C $ es holomorfa entonces para todo $\displaystyle z \in \C $ es
\[f\left(z\right)=\sum^{\infty}_{n=0}\frac{f^{\left(n\right)}\left(0\right)}{n!}z^{n} ,\]
la serie de potencias tiene radio de convergencia infinito. 
\end{observation}
\begin{eg}
La observación anterior se puede usar para deducir que las siguientes igualdades se cumplen uniformemente en todo $\displaystyle \C $, 
\[e^{z}=\sum^{\infty}_{n = 0}\frac{1}{n!}z^{n}, \quad \sin z = \sum^{\infty}_{n = 0}\frac{\left(-1\right)^{n}}{\left(2n+1\right)!}z^{2n+1}, \quad \cos z = \sum^{\infty }_{n = 0}\frac{\left(-1\right)^{n}}{\left(2n\right)!}z^{2n} .\]
\end{eg}
\begin{colorary}
Sean $\displaystyle f,g : \Omega \to \C $ holomorfas, donde $\displaystyle \Omega \subset \C $ es abierto conexo. Supongamos que para algún $\displaystyle z_{0} \in \Omega  $ se tiene que $\displaystyle f\left(z_{0}\right)=g\left(z_{0}\right) $ y que $\displaystyle f^{\left(n\right)}\left(z_{0}\right)=g^{\left(n\right)}\left(z_{0}\right) $ para todo $\displaystyle n\in \N $. Entonces, $\displaystyle f = g $ en $\displaystyle \Omega  $. 
\end{colorary}
\begin{proof}
Considerando la diferencia $\displaystyle f - g $ basta probar el resultado en el caso especial en que $\displaystyle g = 0 $. Consideremos $\displaystyle U \subset \Omega  $ definido por
\[U = \left\{ z \in \Omega \; : \; f\left(z\right) = 0 \; \text{y} \; f^{\left(n\right)}\left(z\right)=0, \; \forall n \in \N\right\}  .\]
Por hipótesis $\displaystyle z_{0}\in U $ por lo que $\displaystyle U \neq \emptyset $. Puesto que $\displaystyle f $ y sus derivadas son continuas, es claro que $\displaystyle U $ es un cerrado relativo en $\displaystyle \Omega  $. Por otro lado, si $\displaystyle w_{0}\in U $, entonces $\displaystyle f\left(w_{0}\right)=f^{\left(n\right)}\left(w_{0}\right)=0 $ para todo $\displaystyle n \in \N $, y aplicando el teorema anterior tenemos que
\[f\left(z\right)=\sum^{\infty}_{n = 0}\frac{f^{\left(n\right)}\left(w_{0}\right)}{n!}\left(z-w_{0}\right)^{n} = 0 ,\; \forall z \in D\left(w_{0}, r\right)\subset \Omega .\]
Se deduce entonces que $\displaystyle f\left(z\right)=f^{\left(n\right)}\left(z\right)=0 $ para todo $\displaystyle z $ en dicho disco y todo $\displaystyle n \in \N $, luego $\displaystyle D\left(w_{0}, r\right)\subset U $. Esto prueba que $\displaystyle U $ es abierto. Como $\displaystyle U $ es abierto no vacío y cerrado relativo a $\displaystyle \Omega  $, que es conexo, debe ser que $\displaystyle U = \Omega  $ por lo que $\displaystyle f = 0 $ en $\displaystyle \Omega  $. 
\end{proof}
\section{Más consecuencias}
\begin{theorem}[Teorema de identidad]
Sea $\displaystyle \Omega \subset \C $ un abierto conexo y $\displaystyle f,g : \Omega \to \C $ holomorfas. Supongamos que existe $\displaystyle E \subset \Omega  $ que tiene algún punto de acumulación dentro de $\displaystyle \Omega  $ y tal que $\displaystyle f = g $ en $\displaystyle E $. Entonces $\displaystyle f = g $ en todo $\displaystyle \Omega  $.
\end{theorem}
\begin{proof}
	Cambiando $\displaystyle f $ por $\displaystyle f-g $ podemos suponer $\displaystyle g = 0 $. También podemos suponer que $\displaystyle E $ es una sucesión $\displaystyle \left\{ z_{n}\right\} _{n\in\N } $ que converge a un punto $\displaystyle z_{0}\in \Omega  $, y tal que $\displaystyle f\left(z_{n}\right)=0 $ y $\displaystyle z_{n} \neq z_{0} $, $\displaystyle \forall n \in \N $. 
	Como $\displaystyle f $ es holomorfa en $\displaystyle \Omega  $, si tomamos $\displaystyle R > 0 $ tal que $\displaystyle \overline{D}\left(z_{0}, R\right)\subset \Omega  $, sabemos que 
	\[f\left(z\right)=\sum^{\infty}_{n = 0}a_{n}\left(z-z_{0}\right)^{n}, \; \forall z \in D\left(z_{0}, R\right) ,\]
	donde $\displaystyle a_{n}=\frac{f^{\left(n\right)}\left(z_{0}\right)}{n!} $, para cada $\displaystyle n \in \N \cup \left\{ 0\right\}  $. Afirmamos que $\displaystyle f\left(z\right)=0 $ para todo $\displaystyle z\in D\left(z_{0}, R\right) $. En efecto, supongamos que $\displaystyle f $ no fuera idénticamente nula en este disco. 
	Entonces eixste algún $\displaystyle n \in \N $ tal que $\displaystyle a_{n} \neq 0 $. Llamemos $\displaystyle m $ al primero de esos números naturales. Entonces,
	\[f\left(z\right)=\sum^{\infty}_{n = m}a_{n}\left(z-z_{0}\right)^{n}=\left(z-z_{0}\right)^{m}\sum^{\infty}_{k = 0}b_{k}\left(z-z_{0}\right)^{k}=\left(z-z_{0}\right)^{m}g\left(z\right) ,\]
	donde $\displaystyle b_{j}=a_{m + j} $ para cada $\displaystyle j \in \N \cup \left\{ 0\right\}  $, y $\displaystyle g\left(z\right)=\sum^{\infty}_{ k = 0}b_{k}\left(z-z_{0}\right)^{k} $. 
	Puesto que $\displaystyle b_{0}=a_{m}\neq 0 $, por continuidad de $\displaystyle g $ existiría $\displaystyle r > 0 $ tal que $\displaystyle g\left(z\right)\neq 0 $, $\displaystyle \forall z \in D\left(z_{0}, r\right) $. Pero entonces
	\[f\left(z\right)=\left(z-z_{0}\right)^{n}g\left(z\right)\neq 0, \; \forall z \in D\left(z_{0}, r\right) \backslash \left\{ z_{0}\right\}  .\]
	Esto contradice que $\displaystyle f\left(z_{n}\right)=0 $ y $\displaystyle z_{n} \neq z_{0} $ para todo $\displaystyle n $, con $\displaystyle \lim_{n \to \infty}z_{n}=z_{0} $. Por tanto, debe ser que $\displaystyle f \equiv 0 $ en $\displaystyle D\left(z_{0}, R\right) $, y como este conjunto es abierto se deduce que todas las derivadas de $\displaystyle f $ se anulan en $\displaystyle z_{0} $, y por supuesto también $\displaystyle f\left(z_{0}\right)=0 $. Un corolario anterior nos dice que $\displaystyle f \equiv 0 $ en todo el abierto conexo $\displaystyle \Omega  $.
\end{proof}
\begin{colorary}[Principio de permanencia de las ecuaciones funcionales]
Sean $\displaystyle \Omega\subset \C $ abierto conexo, $\displaystyle E \subset \Omega  $ con algún punto de acumulación dentro de $\displaystyle \Omega  $, y $\displaystyle F : \Omega \times \Omega \to \C $ tal que las funciones $\displaystyle z \in \Omega \to F\left(z,w\right) $ y $\displaystyle w \in \Omega \to F\left(z,w\right) $ son holomorfas para cada $\displaystyle w \in \Omega  $ y cada $\displaystyle z \in \Omega  $, respectivamente. 
Supongamos que $\displaystyle F\left(z,w\right)=0 $ para todos los $\displaystyle z,w\in E $. Entonces $\displaystyle F\left(z,w\right)=0 $ para todos los $\displaystyle z,w\in \Omega  $.
\end{colorary}
\begin{proof}
Fijemos en primer lugar $\displaystyle z_{0} \in E $. Como $\displaystyle w \to F\left(z_{0}, w\right) $ es holomorfa y se anula en $\displaystyle E $, por el teorema anterior tenemos que $\displaystyle F\left(z_{0}, w\right)=0 $, $\displaystyle \forall w \in \Omega  $. Fijemos ahora $\displaystyle w \in \Omega  $. Sabemos que $\displaystyle F\left(z_{0}, w\right)=0 $, donde $\displaystyle z_{0} \in E $ es arbitrario. 
Es decir, sabemos que $\displaystyle z \to F\left(z,w\right) $ se anula en $\displaystyle E $ y es holomorfa. Por el teorema anterior deducimos que $\displaystyle F\left(z,w\right)=0 $ para todo $\displaystyle z \in \Omega  $. Y como $\displaystyle w \in \Omega  $ es arbitrario, se concluye que $\displaystyle F\left(z,w\right)=0 $, $\displaystyle \forall z,w \in \Omega $. 
\end{proof}
\begin{eg}
Utilizando el teorema anterior se deduce inmediatamente la validez de las igualdades
\[
\begin{split}
 \sin\left(z + w\right)=\sin z \cos w + \cos z \sin w \\
 \cos\left(z + w\right)=\cos z \cos w - \sin z \sin w \\
 \cos ^{2}z + \sin ^{2}z = 1 , \; \forall z,w\in \C.
\end{split}
\]
a partir de las correspondientes igualdades para números reales.
\end{eg}
\begin{theorem}[Teorema de Liouville]
Si $\displaystyle f : \C \to \C $ es holomorfa y acotada entonces $\displaystyle f $ es constante. 
\end{theorem}
\begin{proof}
Puesto que $\displaystyle \C $ es abierto y conexo, basta probar que $\displaystyle f' = 0 $. Esto es consecuencia de las desigualdades de Cauchy: para $\displaystyle z_{0}\in \C $ y $\displaystyle R > 0 $ talesquiera, sabemos que
\[ \left|f^{\left(n\right)}\left(z_{0}\right)\right|\leq \frac{n!}{R^{n}}\sup_{z\in \partial D\left(z_{0}, R\right)} \left|f\left(z\right)\right| .\]
En particular, para $\displaystyle n = 1 $, usando que $\displaystyle f $ está acotada en $\displaystyle \C $, tenemos 
\[ \left|f'\left(z_{0}\right)\right|\leq \frac{\|f\|_{\infty }}{R} ,\]
y haciendo $\displaystyle R \to \infty $ obtenemos que $\displaystyle f'\left(z_{0}\right)=0 $.
\end{proof}
\begin{colorary}[Teorema fundamental del álgebra]
Todo polinomio $\displaystyle P : \C \to \C  $ con coeficientes complejos y de grado $\displaystyle n \in \N $ tiene exactamente $\displaystyle n $ raíces (posiblemente repetidas, contadas con su multiplicidad).
\end{colorary}
\begin{proof}
Supongamos que $\displaystyle P $ no tenga ninguna raíz en $\displaystyle \C $. Entonces
\[f\left(z\right)=\frac{1}{P\left(z\right)}, \; z \in \C ,\]
está bien definida y es holomorfa en $\displaystyle \C $. Es inmediato comprobar que 
\[\lim_{ \left|z\right| \to \infty }f\left(z\right)=0, \]
y junto con la continuidad de $\displaystyle f $ esto implica que $\displaystyle f $ está acotada en $\displaystyle \C $. Pero entonces, por el teorema anterior, $\displaystyle f $ es constante, por lo que $\displaystyle P $ también lo es y esto contradice que $\displaystyle P $ sea de grado mayor o igual que uno. 
Por tanto, $\displaystyle P $ tiene al menos una raíz, llamémosla $\displaystyle z_{1} $. Entonces $\displaystyle P $ se puede factorizar en la forma $\displaystyle P\left(z\right)=\left(z-z_{1}\right)Q_{1}\left(z\right) $, donde $\displaystyle Q_{1} $ es un polinomio de grado $\displaystyle n-1 $. 
Si $\displaystyle n = 1 $ hemos acabado. En caso contrario, aplicamos el argumento anterior para obtener $\displaystyle z_{2}\in \C $ tal que $\displaystyle Q_{1}\left(z_{2}\right)=0 $. En particular, $\displaystyle Q_{1}\left(z\right)=\left(z-z_{2}\right)Q_{2}\left(z\right) $, con $\displaystyle Q_{2} $ es un polinomio de grado $\displaystyle n-2 $. Así,
\[P\left(z\right)=\left(z-z_{1}\right)\left(z-z_{2}\right)Q_{2}\left(z\right) .\]
Reiterando $\displaystyle n-2 $ veces llegamos a que
\[P\left(z\right)=\left(z-z_{1}\right) \cdots \left(z-z_{n}\right)Q_{n} ,\]
donde $\displaystyle Q_{n} $ es una constante. 
\end{proof}
\begin{theorem}[Teorema de Morera]
Sean $\displaystyle \Omega\subset\C $ abierto y $\displaystyle f : \Omega \to \C $ continua. Supongamos que 
\[\int _{T}f =0 ,\]
para todo triángulo $\displaystyle T $ contenido en $\displaystyle \Omega  $ cuya región interior también está contenida en $\displaystyle \Omega  $. Entonces, $\displaystyle f $ es holomorfa en $\displaystyle \Omega  $.
\end{theorem}
\begin{proof}
Supongamos primero que $\displaystyle \Omega  $ es convexo. Entonces tendremos que existe $\displaystyle F : \Omega \to \C $ holomorfa tal que $\displaystyle F' = f $, y como todas las funciones holomorfas tienen derivadas holomorfas debe ser que $\displaystyle f $ es holomorfa. \\
\indent Supongamos ahora que $\displaystyle \Omega  $ no es convexo. Para $\displaystyle z_{0}\in \Omega  $ consideramos un disco abierto $\displaystyle D\left(z_{0}, r\right)\subset \Omega  $. Como este disco es convexo, el argumento anterior prueba que $\displaystyle f $ es holomorfa en $\displaystyle D\left(z_{0}, r\right) $. Repitiendo el mismo argumento para cada punto tenemos que $\displaystyle f $ es holomorfa en $\displaystyle \Omega  $. 
\end{proof}
\begin{theorem}
	Sean $\displaystyle \left\{ f_{n}\right\} _{n\in\N} $ una sucesión de funciones holomorfas en un abierto $\displaystyle \Omega \subset \C $, $\displaystyle f : \Omega \to \C $ una función y supongamos que $\displaystyle f_{n} \to f $ uniformemente en cada subconjunto compacto de $\displaystyle \Omega  $. Entonces, $\displaystyle f $ es holomorfa. 
\end{theorem}
\begin{proof}
Observemos primero que $\displaystyle f $ es continua en $\displaystyle \Omega  $ por ser límite uniforme de funciones continuas en cada subconjunto compacto de $\displaystyle \Omega  $, y en particular en cada disco cerrado contenido en $\displaystyle \Omega  $. Sea $\displaystyle D $ un disco abierto contenido en $\displaystyle \Omega  $ y sea $\displaystyle T $ un triángulo cualquiera contenido en $\displaystyle D $. Como $\displaystyle f_{n} $ es holomorfa en el abierto convexo $\displaystyle D $, por el teorema de Cauchy sabemos que 
\[\int _{T}f_{n} = 0 , \; \forall n \in \N .\]
Entonces, dado $\displaystyle T $ uno cualquiera de estos triángulos, como $\displaystyle \lim_{n \to \infty }f_{n}\left(z\right)=f\left(z\right) $ uniformemente en $\displaystyle z\in T $ se tiene que 
\[\int _{T}f =\int _{T}\lim_{n \to \infty}f_{n} =\lim_{n \to \infty}\int _{T}f_{n} =0 .\]
Por el teorema de Morera tenemos que $\displaystyle f $ es holomorfa en $\displaystyle D $ y como $\displaystyle D $ es arbitrario podemos deducir que $\displaystyle f $ es holomorfa en $\displaystyle \Omega  $. 
\end{proof}
\begin{theorem}
	Con las mismas hipótesis del teorema anterior, se tiene también que la sucesión de derivadas $\displaystyle \left\{ f'_{n}\right\} _{n\in\N} $ converge a $\displaystyle f' $ uniformemente en cada subconjunto compacto de $\displaystyle \Omega  $. En general, para cualquier $\displaystyle j \in \N $ y cualquier compacto $\displaystyle K \subset \Omega  $, la sucesión de derivadas $\displaystyle j $-ésimas $\displaystyle \left\{ f_{n}^{\left(j\right)}\right\} _{n\in\N} $ converge a la derivada $\displaystyle j $-ésima $\displaystyle f^{\left(j\right)} $ uniformemente en $\displaystyle K $.
\end{theorem}
\begin{proof}
Sea $\displaystyle K \subset \Omega  $ compacto y definimos 
\[\delta = \frac{1}{3}d\left(K, \partial \Omega \right) ,\]
si $\displaystyle \Omega \neq \C $ o $\displaystyle \delta = 1 $ si $\displaystyle \Omega = \C $. Entonces el conjunto
\[K_{\delta }: = \left\{ z \in \Omega \; : \; d\left(z,K\right)\leq \delta \right\}  ,\]
sigue siendo compacto y está contenido en $\displaystyle \Omega  $, por lo que $\displaystyle f_{n} \to f $ uniformemente en $\displaystyle K_{\delta } $. 
Para cada $\displaystyle j \in \N $ y cada $\displaystyle z \in K $, tenemos que $\displaystyle \overline{D}\left(z, \delta \right)\subset K_{\delta }\subset \Omega  $ y aplicando las desigualdades de Cauchy a la función holomorfa $\displaystyle f_{n} -f $ obtenemos que 
\[ \left|f_{n}^{\left(j\right)}\left(z\right)-f^{\left(j\right)}\left(z\right)\right| \leq \frac{j!}{\delta ^{j}}\sup_{\xi\in \partial D\left(z, \delta \right)} \left|f_{n}\left(\xi\right)-f\left(\xi\right)\right| \leq \frac{j!}{\delta ^{j}}\sup_{\xi \in K_{\delta }} \left|f_{n}\left(\xi\right)-f\left(\xi\right)\right| .\]
Como el miebro de la derecha de la anterior desigualdad tiende a $\displaystyle 0 $ cuando $\displaystyle n \to \infty $ y no depende de $\displaystyle z \in K $, se concluye el resultado. 
\end{proof}
\begin{colorary}
Si $\displaystyle f_{n} : \Omega \subset \C \to \C $ con $\displaystyle n \in \N $, son funciones holomorfas y la serie $\displaystyle F\left(z\right)=\sum^{\infty}_{n = 1}f_{n}\left(z\right) $ converge uniformemente en cada subconjunto compacto de $\displaystyle \Omega  $, entonces $\displaystyle F : \Omega \to \C $ es holomorfa.
\end{colorary}
\begin{theorem}[Teorema de derivación bajo el signo de la integral]
	Sean $\displaystyle \Omega \subset \C $ abierto, $\displaystyle F : \Omega \times \left[a,b\right] \to \C $ continua tal que $\displaystyle z \in \Omega \to F\left(z,s\right) $ es holomorfa para cada $\displaystyle s \in \left[a,b\right]  $. Entonces la función $\displaystyle f :  \Omega\to \C $ definida por
	\[f\left(z\right)=\int^{b}_{a} F\left(z,s\right) \; d s ,\]
	es holomorfa en $\displaystyle \Omega  $. Además, para cada $\displaystyle z_{0} \in \Omega  $ la función $\displaystyle s \in \left[a,b\right] \to \frac{\partial F}{\partial z}\left(z_{0}, s\right) $ es integrable y se tiene 
	\[f'\left(z_{0}\right)=\int^{b}_{a} \frac{\partial F}{\partial z}\left(z_{0}, s\right) \; d s .\]
\end{theorem}
\begin{proof}
	Haciendo un cambio de variable podemos suponer $\displaystyle \left[a,b\right] = \left[0,1\right]  $. Para cada $\displaystyle n \in \N $ consideramos la suma de Riemann, 
	\[f_{n}\left(z\right)=\frac{1}{n}\sum^{n}_{k = 1}F\left(z, \frac{k }{n}\right) ,\]
	de forma que 
	\[f\left(z\right)= \int^{1}_{0} F\left(z,s\right) \; d s = \lim_{n \to \infty}f_{n}\left(z\right) .\]
Como cada función $\displaystyle z \to F\left(z, \frac{k}{n}\right) $ es holomorfa por hipótesis, está claro que $\displaystyle f_{n} $ es holomorfa para todo $\displaystyle n \in \N $. Por tanto, basta con probar que $\displaystyle f_{n} \to f $ uniformemente en cada disco cerrado contenido en $\displaystyle \Omega  $. 
En efecto, como $\displaystyle F $ es continua y $\displaystyle \overline{D}\left(z_{0}, r\right)\times \left[0,1\right]  $ es compacto, $\displaystyle F $ es uniformemente continua en este conjunto. Así, para cada $\displaystyle \epsilon > 0 $ existe $\displaystyle \delta > 0 $ tal que si $\displaystyle z_{1}, z_{2} \in \overline{D}\left(z_{0}, r\right) $, $\displaystyle s_{1}, s_{2} \in \left[0,1\right]  $ y $\displaystyle \left|z_{1}-z_{2}\right| + \left|s_{1}-s_{2}\right|\leq\delta  $, entonces
\[ \left|F\left(z_{1}, s_{1}\right)-F\left(z_{2}, s_{2}\right)\right|\leq \epsilon  .\]
En particular, 
\[ \left|F\left(z, s_{1}\right)-F\left(z, s_{2}\right)\right|\leq \epsilon, \quad \text{si} \quad z \in \overline{D}\left(z_{0}, r\right), s_{1}, s_{2} \in \left[0,1\right] , \; \text{y} \; \left|s_{1}-s_{2}\right|\leq \delta  .\]
Por tanto, para todo $\displaystyle n \geq \frac{1}{\delta } $, $\displaystyle z \in \overline{D}\left(z_{0}, r\right) $ se tiene 
\[
\begin{split}
	\left|f_{n}\left(z\right)-f\left(z\right)\right| = & \left|\frac{1}{n}\sum^{n}_{k = 1}F\left(z, \frac{k}{n}\right)-\int^{1}_{0} F\left(z,s\right) \; d s\right| = \left|\sum^{n}_{k = 1}\int^{\frac{k}{n}}_{\frac{k-1}{n}} \left(F\left(z, \frac{k}{n}\right)-F\left(z,s\right)\right) \; d s\right| \\
	\leq & \sum^{n}_{k = 1}\int^{\frac{k}{n}}_{\frac{k-1}{n}} \left|F\left(z,\frac{k}{n}\right)-F\left(z,s\right)\right| \; d s \leq \sum^{n}_{k = 1}\frac{\epsilon }{n} = \epsilon .
\end{split}
\]
Esto demuestra la convergencia uniforme de $\displaystyle f_{n} $ a $\displaystyle f $ en $\displaystyle \overline{D}\left(z_{0}, r\right) $, por lo que $\displaystyle f $ es holomorfa en $\displaystyle D\left(z_{0}, r\right) $. Por último, sabemos que la sucesión de derivadas $\displaystyle \left\{ f'_{n}\right\} _{n\in\N} $ también converge a $\displaystyle f' $ uniformemente en $\displaystyle \overline{D}\left(z_{0}, r\right) $. 
En particular, asumiendo que $\displaystyle s \in \left[a,b\right] \to \frac{\partial F}{\partial z}\left(z_{0}, s\right) $ es integrable, tendríamos que 
\[f'\left(z_{0}\right)=\lim_{n \to \infty}f'_{n}\left(z_{0}\right)=\lim_{n \to \infty}\frac{1}{n}\sum^{n}_{k = 1}\frac{\partial F}{\partial z}\left(z_{0}, \frac{k}{n}\right)=\int^{1}_{0} \frac{\partial F}{\partial z}\left(z_{0}, s\right) \; d s .\]
Veamos que esta derivada parcial es integrable Riemann con integral $\displaystyle f'\left(z_{0}\right) $. Usando el teorema de Darboux, bastará ver que para cada $\displaystyle \epsilon > 0 $ existe $\displaystyle \delta > 0 $ tal que para cualquier partición $\displaystyle P \in \mathcal{P}\left(\left[0,1\right] \right) $ en subintervalos de longitud menor o igual que $\displaystyle \delta  $, y para cualesquiera $\displaystyle s_{j}\in \left[t_{j-1}, t_{j}\right]  $ con $\displaystyle j = 1, \ldots, n $, se tiene 
\[ \left|f'\left(z_{0}\right)-\sum^{N}_{j = 1}\frac{\partial F}{\partial z}\left(z_{0}, s_{j}\right)\left(t_{j}-t_{j-1}\right)\right|\leq \epsilon  .\]
Supongamos que esto no fuera así, entonces existiría $\displaystyle \epsilon _{0} > 0 $, una sucesión de particiones $\displaystyle P_{n} = \left\{ t_{n}_{0}, \ldots, t_{N}^{n}\right\} $ de $\displaystyle \left[0,1\right]  $ en subintervalos de longitudes menores que $\displaystyle \frac{1}{n} $ y puntos $\displaystyle s^{n}_{j} $ en cada subintervalo $\displaystyle \left[t ^{n}_{j-1}, t_{j}^{n}\right]  $, $\displaystyle j = 1, \ldots, N_{n} $ tales que 
\[ \left|f'\left(z_{0}\right)-\sum^{N_{n}}_{j = 1}\frac{\partial F}{\partial z}\left(z, s^{n}_{j}\right)\left(t_{j}^{n}-t_{j-1}^{n}\right)\right| > \epsilon  .\]
Para cada $\displaystyle n \in \N $ definimos 
\[g_{n}\left(z\right)= \sum^{N_{n}}_{ j= 1}F\left(z, s^{n}_{j}\right)\left(t_{j}^{n}-t_{j-1}^{n}\right).\]
Cada $\displaystyle g_{n} $ es holomorfa. Dado $\displaystyle \epsilon > 0 $ arbitrario tenemos que si $\displaystyle n > \frac{1}{\delta } $ y $\displaystyle z \in \overline{D}\left(z_{0}, r\right) $ entonces
\[
\begin{split}
	\left|g_{n}\left(z\right)-f\left(z\right)\right|= & \left|\sum^{n}_{k = 1}F\left(z, s_{k}^{n}\right)\left(t_{k}^{n}-t_{k-1}^{n}\right)-\int^{1}_{0} F\left(z,s\right) \; d s \right| = \left|\sum^{N_{n}}_{k = 1}\int^{t_{k}^{n}}_{t_{k - 1}^{n}} \left(F\left(z, s^{n}_{k}\right)-F\left(z,s\right)\right) \; d s\right| \\
	\leq & \sum^{N_{n}}_{k = 1}\int^{t_{k}^{n}}_{t_{k-1}^{n}} \left|F\left(z-s^{n}_{k}\right)-F\left(z,s\right)\right| \; d s \leq \sum^{N_{n}}_{k = 1}\epsilon \left(t_{k}^{n}-t_{k-1}^{n}\right)=\epsilon .
\end{split}
\]
Esto demuestra que $\displaystyle \lim_{n \to \infty}g_{n}\left(z\right)=f\left(z\right) $ uniformemente en $\displaystyle \overline{D}\left(z_{0}, r\right) $. Entonces, tendremos que $\displaystyle g'_{n} $ converge a $\displaystyle f' $ uniformemente en los compactos de $\displaystyle D\left(z_{0}, r\right) $ y en particular $\displaystyle \lim_{n \to \infty}g'\left(z_{0}\right) = f'\left(z_{0}\right) $, lo que contradice una desigualdad anterior. 
\end{proof}

\[g\left(z\right)=\int _{\Gamma }\varphi\left(z, \xi\right) \; d\xi ,\]
es holomorfa. Además, para cada $\displaystyle z_{0} \in \Omega  $ se tiene que 
\[g'\left(z_{0}\right)=\int _{\Gamma }\frac{\partial \varphi}{\partial z}\left(z_{0}, \xi\right) \; d\xi .\]
